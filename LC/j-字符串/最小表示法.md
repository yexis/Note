```cpp
// 字符串最小表示法
string get_min(string& s, int m) {
    int n = s.size();
    int k = 0, i = 0, j = 1;
    while (k < m && i < n && j < n) {
        if (s[(i + k) % n] == s[(j + k) % n]) {
            k++;
        } else {
            if (s[(i + k) % n] > s[(j + k) % n]) i += k + 1;
            else j += k + 1;
            if (i == j) i++;
            k = 0;
        }
    }
    // k是最小表示开头的下标
    k = min(i, j);
    string res;
    for (int i = 0; i < m; i++) res.push_back(s[(k + i) % n]);
    return res;
}
```
