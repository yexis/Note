###### 写法二：hash （单次查询时间复杂度：$\sqrt n$，卡常）

```cpp
using ll = long long;
struct Block {
    vector<ll> a;
    int n;
    int m; // 块个数
    int B; // 块大小
    
    vector<ll> lazy; // 记录完整块操作
    vector<int> bel; // 记录每个节点所属块id

    vector<int> L; // 每个块的左端点
    vector<int> R; // 每个块的右端点

    // 二者选其一
    bool use_umap = false;
    vector<vector<int>> ord; // 每个块内的元素
    vector<unordered_map<ll, int>> mp; // 每个块内的元素

    Block(vector<ll>& aa) {
        a = aa;
        n = a.size();
        B = sqrt(n); 
        m = (n - 1) / B + 1;
        
        lazy.resize(m);
        use_umap = true;

        if (use_umap) {
            mp.resize(m);
            for (int i = 0; i < n - 1; i++) {
                mp[i / B][a[i]]++;
            }
        } else {
            bel.resize(n);
            L.resize(m);
            R.resize(m);
            ord.resize(m);
            for (int id = 0; id < m; id++) {
                L[id] = id * B;
                R[id] = (id + 1) * B - 1;
                for (int i = L[id]; i <= min(n - 1, R[id]); i++) {
                    bel[i] = id;
                }
                rebuild(id);
            }    
        }
        
    }

    void rebuild(int id) {
        ord[id].clear();
        for (int i = L[id]; i <= min(n - 1, R[id]); i++) {
            ord[id].push_back(a[i]);
        }
        sort(ord[id].begin(), ord[id].end());
    }

    void push(int id) {
        if (!lazy[id]) return;
        for (int i = L[id]; i <= min(n - 1, R[id]); i++) {
            a[i] += lazy[id];
        }
        lazy[id] = 0;
        rebuild(id);
    }

    // 区间加
    void add(int l, int r, int v) {
        int lid = l / B;
        int rid = r / B;
        if (use_umap) {
            if (lid == rid) {
                for (int i = l; i <= r; i++) {
                    if (--mp[lid][a[i]] == 0) { 
                        mp[lid].erase(a[i]); 
                    }
                    a[i] += v;
                    ++mp[lid][a[i]];
                }
            } else {
                for (int i = l; i <= min(n - 1, (lid + 1) * B - 1); i++) {
                    if (--mp[lid][a[i]] == 0) { 
                        mp[lid].erase(a[i]); 
                    }
                    a[i] += v;
                    ++mp[lid][a[i]];
                }
                for (int id = lid + 1; id < rid; id++) {
                    lazy[id] += v;
                }
                for (int i = rid * B; i <= min(n - 1, r); i++) {
                    if (--mp[rid][a[i]] == 0) { 
                        mp[rid].erase(a[i]); 
                    }
                    a[i] += v;
                    ++mp[rid][a[i]];
                }
            }
        } else {
            if (lid == rid) {
                push(lid);
                for (int i = l; i <= r; i++) {
                    a[i] += v;
                }
                rebuild(lid);
            } else {
                push(lid);
                for (int i = l; i <= min(n - 1, R[lid]); i++) {
                    a[i] += v;
                }
                rebuild(lid);
    
                for (int id = lid + 1; id < rid; id++) {
                    lazy[id] += v;
                }
    
                push(rid);
                for (int i = L[rid]; i <= r; i++) {
                    a[i] += v;
                }
                rebuild(rid);
            }    
        }
        
    }
    
    // 查询 [l, r] >= K 的个数
    // or 查询 [l, r] <= K 的个数
    // or 查询 [l, r] = K 的个数（需要两次二分）
    // or 查询 [l, r] 第一个大于K
    // or 查询 [l, r] 区间和，最值
    int ask(int l, int r, int K) {
        int ans = 0;
        int lid = l / B;
        int rid = r / B;
        if (use_umap) {
            // 计算块内大于等于v的元素个数
            if (lid == rid) {
                for (int i = l; i <= r; i++) {
                    ans += (a[i] + lazy[lid] == K);
                }
            } else {
                for (int i = l; i <= (lid + 1) * B - 1; i++) {
                    ans += (a[i] + lazy[lid] == K);
                }
                
                for (int id = lid + 1; id < rid; id++) {
                    ans += mp[id][K - lazy[id]];
                }
                
                for (int i = rid * B; i <= r; i++) {
                    ans += (a[i] + lazy[rid] == K);
                }        
                
            }    
        } else {
            // 计算块内大于等于K的元素个数
            if (lid == rid) {
                push(lid);
                for (int i = l; i <= r; i++) {
                    ans += (a[i] == K);
                }
            } else {
                push(lid);
                for (int i = l; i <= R[lid]; i++) {
                    ans += (a[i] >= K);
                }
    
                // 中间部分
                for (int id = lid + 1; id < rid; id++) {
                    int need = K - lazy[id];
                    int curr = lower_bound(ord[id].begin(), ord[id].end(), need) - ord[id].begin();
                    ans += B - curr;
                }
    
                push(rid);
                for (int i = L[rid]; i <= r; i++) {
                    ans += (a[i] >= K);
                }    
            }
        }
        return ans;
    }
};

class Solution {
public:
    using ll = long long;
    vector<int> numberOfPairs(vector<int>& a, vector<int>& nums2, vector<vector<int>>& qs) {
        vector<ll> b; for (int& e : nums2) b.push_back(e);
        int m = a.size(), n = b.size();

        Block bk(b);
        
        vector<int> ans;
        for (auto& q : qs) {
            if (q[0] == 1) {
                int x = q[1], y = q[2], v = q[3];
                bk.add(x, y, v);
            } else if (q[0] == 2) {
                int tot = q[1], t = 0;
                for (auto e : a) {
                    t += bk.ask(0, n - 1, tot - e);
                }
                ans.push_back(t);
            }
        }
        return ans;
    }
};
```







###### 写法二：有序快+ 二分（很容易超时，单次查询时间复杂度: $\sqrt n * log \sqrt n$） 

```cpp
using ll = long long;
struct Block {
    vector<ll> a;
    int n;
    int m; // 块个数
    int B; // 块大小
    
    vector<ll> lazy; // 记录完整块操作
    vector<int> bel; // 记录每个节点所属块id

    vector<int> L; // 每个块的左端点
    vector<int> R; // 每个块的右端点

    vector<vector<int>> ord; // 每个快内的元素
    
    Block(vector<ll>& aa) {
        a = aa;
        n = a.size();
        B = sqrt(n); 
        m = (n - 1) / B + 1;
        
        lazy = vector<int>(m);
        bel.resize(n);

        L.resize(m);
        R.resize(m);

        ord.resize(m);

        for (int id = 0; id < m; id++) {
            L[id] = id * B;
            R[id] = (id + 1) * B - 1;
            for (int i = L[id]; i <= min(n - 1, R[id]); i++) {
                bel[i] = id;
            }
            rebuild(id);
        }
    }

    void rebuild(int id) {
        ord[id].clear();
        for (int i = L[id]; i <= min(n - 1, R[id]); i++) {
            ord[id].push_back(a[i]);
        }
        sort(ord[id].begin(), ord[id].end());
    }

    void push(int id) {
        if (!lazy[id]) return;
        for (int i = L[id]; i <= min(n - 1, R[id]); i++) {
            a[i] += lazy[id];
        }
        lazy[id] = 0;
        rebuild(id);
    }

    // 区间加
    void add(int l, int r, int v) {
        int lid = bel[l];
        int rid = bel[r];
        if (lid == rid) {
            push(lid);
            for (int i = l; i <= r; i++) {
                a[i] += v;
            }
            rebuild(lid);
        } else {
            push(lid);
            for (int i = l; i <= min(n - 1, R[lid]); i++) {
                a[i] += v;
            }
            rebuild(lid);

            for (int id = lid + 1; id < rid; id++) {
                lazy[id] += v;
            }

            push(rid);
            for (int i = L[rid]; i <= r; i++) {
                a[i] += v;
            }
            rebuild(rid);
        }
    }
    
    // 查询 [l, r] >= K 的个数
    // or 查询 [l, r] <= K 的个数
    // or 查询 [l, r] = K 的个数（需要两次二分）
    // or 查询 [l, r] 第一个大于K
    // or 查询 [l, r] 区间和，最值
    int ask(int l, int r, int K) {
        int ans = 0;
        int lid = bel[l];
        int rid = bel[r];
        if (lid == rid) {
            push(lid);
            for (int i = l; i <= r; i++) {
                ans += (a[i] >=  K);
            }
        } else {
            push(lid);
            for (int i = l; i <= R[lid]; i++) {
                ans += (a[i] >= K);
            }

            // 中间部分
            for (int id = lid + 1; id < rid; id++) {
                int need = K - lazy[id];
                int curr = lower_bound(ord[id].begin(), ord[id].end(), need) - ord[id].begin();
                ans += B - curr;
            }

            push(rid);
            for (int i = L[rid]; i <= r; i++) {
                ans += (a[i] >= K);
            }
        }
        return ans;
    }
};

class Solution {
public:
    using ll = long long;
    vector<int> numberOfPairs(vector<int>& a, vector<int>& nums2, vector<vector<int>>& qs) {
        vector<ll> b; for (int& e : nums2) b.push_back(e);
        int m = a.size(), n = b.size();

        Block bk(b);
        
        vector<int> ans;
        for (auto& q : qs) {
            if (q[0] == 1) {
                int x = q[1], y = q[2], v = q[3];
                bk.add(x, y, v);
            } else if (q[0] == 2) {
                int tot = q[1], t = 0;
                for (auto e : a) {
                    t += bk.ask(0, n - 1, tot - e) - bk.ask(0, n - 1, tot - e + 1);
                }
                ans.push_back(t);
            }
        }
        return ans;
    }
};
```
