// 区间定义
struct Query {
    int bid; // 块id
    int l;
    int r;   // 左闭右闭
    int threshold;
    int qid; // 询问id
    bool operator<(const auto& that) const {
        if (bid == that.bid) {
            return r < that.r;
        }
        return bid < that.bid;
    }
};
class Solution {
public:
    using ll = long long;
    vector<bool> validSubarrays(vector<int>& a, int K, vector<vector<int>>& qs) {
        int n = a.size(), m = qs.size();

        vector<int> mp(100010);
        // odd:出现次数是奇数的元素个数
        // sum:出现的元素个数
        ll odd = 0, sum = 0; 
        auto add = [&](ll x) -> void {
            mp[x]++;
            if (mp[x] & 1) odd++;
            else odd--;
            if (mp[x] == 1) sum++;
        };
        auto rem = [&](ll x) -> void {
            mp[x]--;
            if (mp[x] & 1) odd++;
            else odd--;
            if (mp[x] == 0) sum--;
        };     
        
        vector<bool> ans(m);
        int block_size = ceil(n / sqrt(m * 2));
        
        vector<Query> pr;
        for (int i = 0; i < m; i++) {
            auto&q = qs[i];
            int l = q[0], r = q[1];
            // 大区间：离线
            if (r - l + 1 > block_size) {
                pr.emplace_back(l / block_size, l, r, 0, i);
                continue;
            }
            // 小区间：暴力
            int odd = 0, sum = 0;
            unordered_map<int, int> mp;
            for (int j = l; j <= r; j++) {
                int x = a[j];
                if (++mp[x] == 1) sum++;
                if (mp[x] & 1) odd++;
                else odd--;
            }
            if (sum == K && odd == 0) ans[i] = true;
        }

        // 右端点：从l0(包含)只能向右单向移动
        // 左端点：从l0(不包含)只能向左移动，每次回滚回到l0
        sort(pr.begin(), pr.end());

        int r;
        for (int i = 0; i < pr.size(); i++) {
            auto& q = pr[i];
            // l0: 右端点的起点 设置为当前块右边界的下一个点
            int l0 = (q.bid + 1) * block_size;
            // 遍历到新的块，这里最多执行 sqrt(n)次，所以内部可以是O(n)的
            if (i == 0 || q.bid > pr[i - 1].bid) {
                r = l0;
                fill(mp.begin(), mp.end(), 0);
                odd = 0, sum = 0;
            }

          	// 对于大区间
            // 右端点只能向右移动，从 r 移动到 q.r
            for (; r <= q.r; r++) {
                add(a[r]);
            }
            
            // 左端点从 l0(不包含) 移动到 q.l
          	// 左端点起点在当前块的右端点，从右往左移
            for (int j = q.l; j < l0; j++) {
                add(a[j]);
            }
            if (sum == K && odd == 0) ans[q.qid] = true;
            // 回滚：左端点回到当前块的右边界的下一个点，即l0
            for (int j = q.l; j < l0; j++) {
                rem(a[j]);
            }
        }
        // =========== 回滚莫队 模板结束
        
        return ans;
        
    }
};
