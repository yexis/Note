### Title 分块

##### 1 emphasis

- 分块
- 区间加 + 单点询问元素个数

##### 3 key points

令数组存在$n$个元素，分块大小为$B$

 分个块关键因素：

- 块的个数：$(n - 1) / B + 1$（索引从0开始）
- 下标$i$所处区间编号：$i / B$
- 下标$i$所处区间左端点：$i / B * B$
- 下标$i$所处区间右端点：$(i / B + 1) * B  - 1$

> 例题：[递增后的数对数量](https://leetcode.cn/contest/weekly-contest-503/problems/number-of-pairs-after-increment/description/)

```cpp
class Solution {
public:
    using ll = long long;
    vector<int> numberOfPairs(vector<int>& a, vector<int>& nums2, vector<vector<int>>& qs) {
        vector<ll> b; for (int e : nums2) b.push_back(e);
        int m = a.size(), n = b.size();
        
        // 分块思想
        int B = sqrt(n);
        // (n - 1) / B 最后一个元素所在的区间
        int L = (n - 1) / B + 1;
        ll delta[L]; memset(delta, 0, sizeof(delta));
        unordered_map<int, int> mp[L];
        for (int i = 0; i < n; i++) mp[i / B][b[i]]++;
        
        // 单个块内进行add O(sqrt(n))
        auto add = [&](int l, int r, int v) -> void {
            int id = l / B;
            for (int i = l; i <= r; i++) {
                int t = --mp[id][b[i]];
                if (t == 0) mp[id].erase(b[i]);
                b[i] += v;
                ++mp[id][b[i]];
            }
        };
        

        vector<int> ans;
        for (auto& q : qs) {
            if (q[0] == 1) {
                int x = q[1], y = q[2], v = q[3];
                int id1 = x / B, id2 = y / B;
                if (id1 == id2) {
                    add(x, y, v);
                } else {
                    add(x, (id1 + 1) * B - 1, v);
                    add(id2 * B, y, v);
                    for (int j = id1 + 1; j < id2; j++) delta[j] += v;
                }
            } else if (q[0] == 2) {
                int tot = q[1], t = 0;
                for (ll e : a) {
                    for (int j = 0; j < L; j++) {
                        long long goal = tot - delta[j] - e;
                        if (mp[j].count(goal)) t += mp[j][goal];
                    }    
                }
                ans.push_back(t);
            }
        }
        return ans;
    }
};©leetcode
```
