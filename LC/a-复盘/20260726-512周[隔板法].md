// 1 1 | 1 1 1 | 1 | 1 1 1
    
// 将元素x，分解成k个正整数，有多少种方法
// 答案：使用隔板法，即在n-1个缝中插入k-1个隔板 C_{n-1}^{k-1}

// 将元素x，分解成k个奇数，有多少种方法
// 答案：
// 要求每个数是奇数，无法直接使用隔板法
// 每个数ai是奇数，则令ai=2*bi+1，有sum(2*bi+1) = n
// 2*bi = n - k, 则bi = (n - k) / 2
// 此时，bi可以是任意正整数，不再受必须是奇数的条件限制，但是bi可以是0
// n-k是奇数，则无解
// n-k是偶数，则使用隔板法，答案是 C_{(n-k)/2}^{k-1}
using ll = long long;
static constexpr int mod = 1e9 + 7;
ll qpow(ll n, ll b) {
    ll ans = 1;
    while (b) {
        if (b & 1) ans *= n, ans %= mod;
        n *= n, n %= mod;
        b >>= 1;
    }
    return ans;
}
struct Comb {
    ll fac[500010], inv[500010];
    Comb(int n) {
        fac[0] = inv[0] = 1;
        for (int i = 1; i <= n; i++) {
            fac[i] = fac[i - 1] * i; fac[i] %= mod;
            inv[i] = get_inv(fac[i]);
        }
    }
    ll get_inv(ll x) {
        return qpow(x, mod - 2);
    }
    ll C(int n, int m) {
        if (n < 0 || m < 0 || n < m) return -1;
        return fac[n] * inv[m] % mod * inv[n - m] % mod;
    }
};
class Solution {
public:
    // 序列中存在一个偶数，则乘积是偶数
    // 那么答案 = 所有方案数 - 序列全部是奇数的方案数
    int countValidSequences(int n, int K) {
        Comb comb(n);
        ll all = comb.C(n - 1, K - 1);
        if ((n - K) & 1) return all;
        
        int rst = (n - K) >> 1;
        ll odd = comb.C(rst + K - 1, K - 1);
        return (all - odd + mod) % mod;
    }
};
