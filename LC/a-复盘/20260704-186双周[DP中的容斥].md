**方法一：DP中的容斥**（错误代码）

```cpp
class Solution {
public:
    using ll = long long;
    const int mod = 1e9 + 7;
    int interleaveCharacters(string s1, string s2, string target) {
        int m = s1.size(), n = s2.size(), K = target.size();
        if (K == 1) return 0;

        ll dp[m + 1][2][n + 1][2][K + 1];
        for (int i = 0; i <= m; i++) {
            for (int d1 = 0; d1 < 2; d1++) {
                for (int j = 0; j <= n; j++) {
                    for (int d2 = 0; d2 < 2; d2++) {
                        for (int k = 0; k <= K; k++) {
                            dp[i][d1][j][d2][k] = -1;
                        }
                    }
                }
            }
        }

        auto mod_add = [&](ll& x, ll y) -> void {
            x += y; x %= mod; return;
        };
        
        auto dfs = [&](auto&& dfs, int i, int d1, int j, int d2, int k) -> ll {
            if (k == K) {
                if (d1 && d2) return 1;
                return 0;
            }
            if (i == m && j == n) return 0;
            if (dp[i][d1][j][d2][k] != -1) return dp[i][d1][j][d2][k];

            ll& ans = dp[i][d1][j][d2][k];  ans = 0;
            
            // 不选
            // 这里直接跳过了两个字符串的当前字符，肯定是不对的
            // 需要单独考虑跳过 s1[i] 和 s2[j]
            // 但是如果写成两个单独跳过，这里会存在重复计算的问题
            mod_add(ans, dfs(dfs, min(m, i + 1), d1, j, d2, k));
            mod_add(ans, dfs(dfs, i, d1, min(n, j + 1), d2, k));
            mod_add(ans, -dfs(dfs, min(m, i + 1), d1, min(n, j + 1), d2, k));

            // 选
            if (i < m && s1[i] == target[k] && j < n && s2[j] == target[k]) {
                mod_add(ans, dfs(dfs, i + 1, 1, j, d2, k + 1) );
                mod_add(ans, dfs(dfs, i, d1, j + 1, 1, k + 1) );
            } else if (i < m && s1[i] == target[k]) {
                mod_add(ans, dfs(dfs, i + 1, 1, j, d2, k + 1) );
            } else if (j < n && s2[j] == target[k]) {
                mod_add(ans, dfs(dfs, i, d1, j + 1, 1, k + 1) );
            }
            
            return ans;
        };
       
        ll ans = dfs(dfs, 0, 0, 0, 0, 0);
        return ans;
    }
};
```





**方法二：枚举选哪个（枚举选或不选会重复）**（正确代码）

前缀和优化DP

```cpp
class Solution {
public:
    static constexpr int MOD = 1e9 + 7;

    int interleaveCharacters(string word1, string word2, string target) {
        int n = word1.size();
        int m = word2.size();

        vector<vector<int>> dp(n + 1, vector<int>(m + 1));
        dp[0][0] = 1;

        for (char c : target) {
            vector<vector<int>> ndp(n + 1, vector<int>(m + 1));

            // 前缀和数组
            vector<int> row(n + 1, 0);
            vector<int> col(m + 1, 0);

            for (int i = 0; i <= n; i++) {
                for (int j = 0; j <= m; j++) {
                    // word2 方向的前缀和
                    row[i] = (row[i] + dp[i][j]) % MOD;
                    // word1 方向的前缀和
                    col[j] = (col[j] + dp[i][j]) % MOD;

                    // 从 word1[i] 选择当前字符
                    if (i < n && word1[i] == c) {
                        ndp[i + 1][j] += col[j]; ndp[i + 1][j] %= MOD;
                    }

                    // 从 word2[j] 选择当前字符
                    if (j < m && word2[j] == c) {
                        ndp[i][j + 1] += row[i]; ndp[i][j + 1] %= MOD;
                    }
                }
            }

            dp = move(ndp);
        }

        int ans = 0;

        // word1 和 word2 都至少使用一次
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                ans += dp[i][j]; ans %= MOD;
            }
        }

        return ans;
    }
};
```



**方法三：添加f控制target[k]当前必须从哪个字符串取**

**对于状态 `dp[i][j][k]`，当前 `target[k]` 既可以从 `s1[i]` 后面选择，也可以从 `s2[j]` 后面选择。若直接同时对两个字符串进行“选/不选”，会产生跳过顺序不同但实际方案相同的重复路径。**

**因此增加状态 `f`，强制规定当前 `target[k]` 必须从哪个字符串中选择。这样对于当前状态，实际上只存在一个候选字符串，可以重新使用经典的“选/不选”递推。当前字符被选中后，再让 `f` 在两个字符串之间自由选择，从而决定下一个 `target[k+1]` 从哪个字符串获取。**

```cpp
class Solution {
public:
    using ll = long long;
    const int mod = 1e9 + 7;
    ll power(ll x, ll b) {
        ll ans = 1;
        while (b) {
            if (b & 1) ans *= x, ans %= mod;
            x *= x, x %= mod;
            b >>= 1;
        }
        return ans;
    }

    int interleaveCharacters(string s1, string s2, string target) {
        int m = s1.size(), n = s2.size(), K = target.size();
        if (K == 1) return 0;

        ll dp[m + 1][2][n + 1][2][2][K + 1];
        for (int i = 0; i <= m; i++) {
            for (int d1 = 0; d1 < 2; d1++) {
                for (int j = 0; j <= n; j++) {
                    for (int d2 = 0; d2 < 2; d2++) {
                        for (int f = 0; f < 2; f++) {
                            for (int k = 0; k <= K; k++) {
                                dp[i][d1][j][d2][f][k] = -1;
                            }
                        }
                    }
                }
            }
        }

        auto mod_add = [&](ll& x, ll y) -> void {
            x += y; x %= mod; return;
        };
        
        // f表示当前从哪个字符串选择，且必须选
        // 虽然可以跳过当前字符，但跳过时，f不能变
        // 其实整体思路还是考虑当前target[k]是从s1取，还是从s2取
        auto dfs = [&](auto&& dfs, int i, int d1, int j, int d2, int f, int k) -> ll {
            if (k == K) {
                if (d1 && d2) return 1;
                return 0;
            }
            if (i == m && j == n) return 0;
            if (dp[i][d1][j][d2][f][k] != -1) return dp[i][d1][j][d2][f][k];

            ll& ans = dp[i][d1][j][d2][f][k];  ans = 0;

            // 不选
            if (!f) {
                mod_add(ans, dfs(dfs, min(m, i + 1), d1, j, d2, f, k));
            } else {
                mod_add(ans, dfs(dfs, i, d1, min(n, j + 1), d2, f, k));
            }

            // 选
            if (!f) {
                if (i < m && s1[i] == target[k]) {
                    mod_add(ans, dfs(dfs, i + 1, 1, j, d2, 0, k + 1));
                    mod_add(ans, dfs(dfs, i + 1, 1, j, d2, 1, k + 1));
                }
            } else {
                if (j < n && s2[j] == target[k]) {
                    mod_add(ans, dfs(dfs, i, d1, j + 1, 1, 0, k + 1));
                    mod_add(ans, dfs(dfs, i, d1, j + 1, 1, 1, k + 1));
                }
            }

            return ans;
        };
       
        ll ans = 0;
        ans += dfs(dfs, 0, 0, 0, 0, 0, 0); ans %= mod;
        ans += dfs(dfs, 0, 0, 0, 0, 1, 0); ans %= mod;
        ans *= power(2, mod - 2); ans %= mod;
        return ans;
    }
};
```
