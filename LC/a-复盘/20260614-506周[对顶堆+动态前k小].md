### T4 [3962. 至多 K 次交换后最大子数组和](https://leetcode.cn/problems/maximum-subarray-sum-after-at-most-k-swaps/)

##### 1 emphasis

- 对顶堆



##### 2 key points



##### 3 thought

固定一个子数组 [i,j]，考虑交换能带来什么：每次交换可以把窗口内的一个元素换成窗口外的一个元素。要让窗口和最大，显然应该把窗口内最小的若干元素，换成窗口外最大的若干元素，只要换进来的比换出去的大就值得换。

如果对每个窗口都重新排序求最小/最大的若干个，复杂度太高。注意固定左端点 i，右端点 j 从左往右扩展时，有两个性质：每次只是把一个元素从窗口外移到窗口内。



**假设目前的最优方案是交换 t 对元素，当把一个元素从窗口外移动到窗口内时，t 只会变化 1。请读者仔细思考这是为什么。**
**于是我们用两个对顶堆 inner 和 outer 动态维护窗口内和窗口外的前 t 大元素之和，每次把一个元素加入 inner，然后从 outer 移除一个元素，再看 t 是否要改变 1。每次这种操作的复杂度为 O(logn)。整体复杂度 $O(n^2 log(n))$。**



**K每次至多变化1**



本题由于常数比较紧张，参考代码将对顶堆的模板从使用 multiset 变成了使用懒删除堆。



##### 4 code

```cpp
// 对顶堆，求前k小元素和 | 第k小的元素
using ll = long long;
struct Magic {
    int K;
    priority_queue<int> pq1, del1; // 最大堆维护前k小
    priority_queue<int, vector<int>, greater<int>> pq2, del2; // 最小堆维护前k小外的部分
    int sz1, sz2; // 懒删除堆，需要实时维护两个堆的元素数量
    ll sm1;
    Magic() {
        K = 0;
        sz1 = sz2 = 0;
        sm1 = 0;
    }
    
    int top1() {
        // 需保证pq1不为空
        while (!del1.empty() && del1.top() == pq1.top()) del1.pop(), pq1.pop();
        return pq1.top();
    }
    int top2() {
        // 需保证pq2不为空
        while (!del2.empty() && del2.top() == pq2.top()) del2.pop(), pq2.pop();
        return pq2.top();
    }
    void adjust() {
        while (sz2 > 0 && sz1 < K) {
            int x = top2();
            pq2.pop(); sz2--;
            pq1.push(x); sz1++; sm1 += x;
        }
        while (sz1 > K) {
            int x = top1();
            pq1.pop(); sz1--; sm1 -= x;
            pq2.push(x); sz2++;
        }
    }
    void add(int x) {
        if (sz2 > 0 && x >= top2()) pq2.push(x), sz2++;
        else pq1.push(x), sz1++, sm1 += x;
        adjust();
    }
    void del(int x) {
        if (sz1 > 0 && x <= top1()) del1.push(x), sz1--, sm1 -= x;
        else del2.push(x), sz2--;
        adjust();
    }
};
// 对顶堆模板结束

class Solution {
public:
    long long maxSum(vector<int>& a, int K) {
        int n = a.size();
        ll ans = -1e18;
        for (int left = 0; left < n; left++) {
            // inner 维护窗口内的最小的K个元素
            // outer 维护窗口外的最大的K个元素，使用最小堆维护相反数
            Magic inner, outer;
            for (int j = 0; j < n; j++) outer.add(-a[j]);
            ll sm = 0;
            for (int right = left; right < n; right++) {
                // sz 当前可更换的最大值，和K同时生效
                int sz = right - left + 1;
                sz = min(sz, n - sz);

                sm += a[right];
                inner.add(a[right]), outer.del(-a[right]);
                
                // 交换次数不能超过窗口内/外的元素数量
                while (inner.K > sz) {
                    inner.K--; inner.adjust();
                    outer.K--; outer.adjust();
                }

                // 交换数量是否可能+1
                // inner.K < sz，说明对顶堆中一定存在多余的元素
                while (inner.K < K && inner.K < sz && inner.top2() < -outer.top2()) {
                    inner.K++; inner.adjust();
                    outer.K++; outer.adjust();
                }

                // 交互数量是否需要-1
                while (inner.K > 0 && inner.top1() >= -outer.top1()) {
                    inner.K--; inner.adjust();
                    outer.K--; outer.adjust();
                }
                ans = max(ans, sm - inner.sm1 - outer.sm1);
            }
        }
        return ans;
    }
};
```
