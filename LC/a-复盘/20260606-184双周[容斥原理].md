### T4 [Q4. 互质元素的最大得分](https://leetcode.cn/contest/biweekly-contest-184/problems/maximum-score-with-co-prime-element/)

##### 1 emphasis

- 快速求出数组中与$x$互质的元素
- 快速求出数组中与$x$不互质的元素



##### 2 key points



##### 3 thought

问题：如何快速询问数组$a$中存在多少个数与$x$不互质

> **枚举$x$的所有因子，考虑每个因子被数组$a$中多少个元素整除**
>
> 对于每个因子$p$，减去能整除$p$的元素个数
>
> 但是这样会导致重复，比如 元素$6$，会被因子$2$减去一次，会被因子$3$减去一次
>
> 所以需要采用容斥原理，排除重复的部分（采用二进制状态枚举所有因子，奇数置位减，偶数置位加）



##### 4 code

```cpp

static constexpr int N = 1e5;
vector<int> F[N + 1];
vector<int> prime;
bool not_prime[N + 1];
// smallest prime factor
int spf[N + 1];
// 获取所有质因数 O(log(V))
vector<int> get_all_spf(int x) {
    vector<int> pfs;
    int xx = x;
    while (xx > 1) {
        int spfx = spf[xx];
        int y = x / spfx;
        pfs.push_back(spfx);
      	// 消磨掉所有当前质因子spfx
        while (xx % spfx == 0) xx /= spfx;
    }
    return pfs;
}

auto init = []() -> int {
  	not_prime[0] = not_prime[1] = true;
    for (int i = 2; i <= N; ++i) {
        if (!not_prime[i]) {
            prime.push_back(i);
            spf[i] = i; // 质数的最小质因数是自己
        }
        for (int p : prime) {
          	int m = i * p;
            if (m > N) break;
            not_prime[m] = true;
            spf[m] = p;
            if (i % p == 0) {
                // i % p == 0
                // 换言之，i 之前被 p 筛过了
                // 由于 prime 里面质数是从小到大的，所以 i 乘上其他的质数的结果 一定会被 p 筛掉
                // 就不需要在这里先筛一次，所以这里直接 break 掉就好了
              	// 证明：
              	// 假设 i / p = k，设 q 为质数，且 q > p 
              	// 那么，m = i * q = p * (k * q)，所以m一定会被 p 和 (k*q) 筛掉
                break;
            }
        }
    }
    for (int i = 1; i <= N; i++) F[i] = get_all_spf(i);
    
    return 0;
}();


class Solution {
public:
    int maxScore(vector<int>& a, int maxVal) {
        int n = a.size();

        // 2^6 = 64
        // cout << 2 * 3 * 5 * 7 * 11 * 13 * 15 << "\n";

        unordered_map<int, int> cnt;
        for (int i = 0; i < n; i++) {
            int M = F[a[i]].size();
            for (int mask = 1; mask < (1 << M); mask++) {
                int v = 1;
                for (int d = 0; d < M; d++) {
                    if (mask >> d & 1) v *= F[a[i]][d];
                }
                cnt[v]++;
            }
        }
        
        int ans = -1e9;
        for (int i = 0; i < n; i++) {
            int x = a[i];
            // 自己不变
            int curr = 0;
            auto& fas = F[x];
            int M = fas.size();
            for (int mask = 1; mask < (1 << M); mask++) {
                int l, v; l = 0, v = 1;
                for (int d = 0; d < M; d++) {
                    if (mask >> d & 1) l++, v *= fas[d];
                }
                if (l & 1) {
                    curr += cnt[v];
                    if (x % v == 0) curr--;
                } else {
                    curr -= cnt[v];
                    if (x % v == 0) curr++;
                }
            }   
            ans = max(ans, x - curr);

            // 从maxVal开始
            curr = 0;
            int y = maxVal; if (y == x) y--;
            while (y > 0) {
                int curr = 0;
                auto& fas = F[y];
                int M = fas.size();
                for (int mask = 1; mask < (1 << M); mask++) {
                    int l, v; l = 0, v = 1;
                    for (int d = 0; d < M; d++) {
                        if (mask >> d & 1) l++, v *= fas[d];
                    }
                    if (l & 1) {
                        curr += cnt[v];
                        if (x % v == 0) curr--;
                    } else {
                        curr -= cnt[v];
                        if (x % v == 0) curr++;
                    }
                }
                ans = max(ans, y - curr - 1);
                if (!not_prime[y]) break;
                y--;
            }
        }
        return ans;
    }
};
```
