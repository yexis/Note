### T4 [Q4. 最多有效分割位置 II](https://leetcode.cn/contest/biweekly-contest-190/problems/maximum-valid-split-positions-ii/)

##### 1 emphasis

- gcd的性质
- logtrick



##### 2 key points

**区间gcd性质:**

- 固定左端点时，区间越长，gcd越小（具有单调性）
- 固定左端点时，gcd的值是梯度
- 固定左端点时，相同gcd的值形成的区间右端点一定是连续的，比如 $gcd = [20,20,20,10,10,5,5,5,1]$，所以可以采用二分



##### 3 thought

- 题目要求 $gcd(0,...,i] = gcd(i + 1,..., n - 1)$，那么 $gcd(0,...,i] = gcd(i + 1,..., n - 1) = gcd(数组)$，

/*
 * ands: 记录子数组的的按位与结果 以及拥有相同值子数组右端点的最小值

static constexpr int maxn = 100010;
struct RMQ {
    int Log;
    int n;
    vector<vector<int>> st;
    RMQ(vector<int>& ob) {
        n = ob.size();
        // Log 存储n以内最大的二次幂指数，实在不行可以设置成最大 32
        Log = 32 - __builtin_clz(n);
        // 这里一定不能使用resize，resize不会更新已有范围
        st.assign(n, vector<int>(Log, -1));
        for (int i = 0; i < n; i++) {
            st[i][0] = ob[i];
        }

        for (int d = 1;  d < Log; d++) {
            for (int i = 0; i + (1 << (d - 1)) < n; i++) {
                st[i][d] = gcd(st[i][d - 1], st[i + (1 << (d - 1))][d - 1]);
            }
        }
    }
    int ask(int l, int r) {
        if (l > r) {
            // 这里最好返回0
            // 方便和其他元素gcd
            return 0;
        }
        int d = 31 - __builtin_clz(r - l + 1);
        return gcd(st[l][d], st[r - (1 << d) + 1][d]);
    }
};

vector<pii> cal(vector<int>& nums) {
    vector<pii> ans;
    int n = nums.size();

    for (int i = n - 1; i >= 0; i--) {
        int num = nums[i];
        ans.emplace_back(num, i);

        int k = 0;
        for (int j = 0; j < ans.size(); j++) {
            ans[j].first = gcd(ans[j].first, num);
            if (ans[k].first == ans[j].first) {
                ans[k].second = ans[j].second;
            } else {
                ans[++k] = ans[j];
            }
        }
        ans. resize(k + 1);
    }
    return ans;
}

class Solution {
public:
    int maxValidSplits(vector<int>& a) {
        int n = a.size();
        if (n == 1) return 0;

        vector<int> L(n), R(n);
        for (int i = 0; i < n; i++) {
            if (i == 0) L[i] = a[i];
            else L[i] = gcd(L[i - 1], a[i]);
        }
        for (int i = n - 1; i >= 0; i--) {
            if (i == n - 1) R[i] = a[i];
            else R[i] = gcd(R[i + 1], a[i]);
        }

        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            if (L[i] == R[i + 1]) ans++;
        }
        
        auto trl = cal(a);
        // trl: 
        // 0->, 2->, ...
        
        auto b = a; reverse(b.begin(), b.end());
        auto trr = cal(b);
        // trr:
        // ..., <-3, <-n-1

        RMQ rmq(a);
        
        auto calc = [&](int del) -> int {
            int all = gcd(rmq.ask(0, del - 1), rmq.ask(del + 1, n - 1));

            // 计算前缀gcd，包含pos
            auto gcdPrefix = [&](int pos) -> int {
                if (pos < del) return rmq.ask(0, pos);
                else return gcd(rmq.ask(0, del - 1), rmq.ask(del + 1, pos));
            };
            // 计算后缀gcd，包含pos
            auto gcdSuffix = [&](int pos) -> int {
                if (pos <= del) return gcd(rmq.ask(pos, del - 1), rmq.ask(del + 1, n - 1));
                else return rmq.ask(pos, n - 1);
            };

            // L 第一个大于等于all的位置（一定存在）
            auto check = [&](int mid) -> bool {
                return gcdPrefix(mid) == all;  
            };
            // R 第一个等于等于all的位置（一定存在）
            auto check2 = [&](int mid) -> bool {
                return gcdSuffix(mid) == all;
            };

            int l = 0, r = n - 1, ans = -1;
            while (l <= r) { 
                int mid = (l + r) >> 1;
                if (check(mid)) {
                    ans = mid;
                    r = mid - 1;
                } else {
                    l = mid + 1;
                }
            }
            int start = ans;

            l = 0, r = n - 1, ans = -1;
            while (l <= r) {
                int mid = (l + r) >> 1;
                if (check2(mid)) {
                    ans = mid;
                    l = mid + 1;
                } else {
                    r = mid - 1;
                }
            }
            int end = ans;

            int cnt = end - start;
            if (del >= start && del <= end) cnt--;
            return cnt;
        };

        // 删除i
        for (int i = 0; i < n; i++) {
            int tmp = calc(i);    
            ans = max(ans, tmp);
        }

        return ans;
    }
};

vector<pii> logtrick(vector<int>& nums) {
    vector<pii> ans;
    int n = nums.size();

    for (int i = n - 1; i >= 0; i--) {
        int num = nums[i];
        ans.emplace_back(num, i);

        int k = 0;
        for (int j = 0; j < ans.size(); j++) {
            ans[j].first = gcd(ans[j].first, num);
            if (ans[k].first == ans[j].first) {
                ans[k].second = ans[j].second;
            } else {
                ans[++k] = ans[j];
            }
        }
        ans. resize(k + 1);
    }
    return ans;
}

class Solution {
public:
    int maxValidSplits(vector<int>& a) {
        int n = a.size();
        if (n == 1) return 0;

        int ans = 0;
        auto calc = [&](vector<int>& b) -> int {
            int m = b.size();
            vector<int> L(m), R(m);
            for (int i = 0 ;i < m; i++) {
                if (i == 0) L[i] = b[i];
                else L[i] = gcd(L[i - 1], b[i]);
            }
            for (int i = m - 1; i >= 0; i--) {
                if (i == m - 1) R[i] = b[i];
                else R[i] = gcd(R[i + 1], b[i]);
            }
            int ans = 0;
            for (int i = 0; i < m - 1; i++) {
                if (L[i] == R[i + 1]) ans++;
            }
            return ans;
        };

        ans = calc(a);

        // trick
        // trl: 0->, 2->, ...
        // 记录前缀gcd
        auto trl = logtrick(a);
        
        // trr: ..., <-3, <-n-1
        // 记录后缀gcd
        auto b = a; reverse(b.begin(), b.end());
        auto trr = logtrick(b);
        

        // 对于下标i，如果 L[i] == L[i - 1] && R[i] == R[i + 1]
        // 那么说明下标i 是否删除 对结果不会造成影响
        // 反之，L[i] != L[i - 1] || R[i] != R[i + 1]时 删除下标i 才有可能影响答案
        // 所以，只有gcd发生变化的位置，才有可能影响答案，而这些位置只有log(U)个
        for (auto& [v, pos] : trl) {
            vector<int> c;
            for (int i = 0; i < n; i++) {
                if (i != pos) c.push_back(a[i]);
            }
            int tmp = calc(c);
            ans = max(ans, tmp);
        }
        for (auto& [v, pos] : trr) {
            vector<int> c;
            // 由于是将a翻转后计算的trick
            // 所以翻转后的下标pos，实际上下标是 n - 1 - pos
            for (int i = 0; i < n; i++) {
                if (i != n - 1 - pos) c.push_back(a[i]);
            }
            int tmp = calc(c);
            ans = max(ans, tmp);
        }
        return ans;
    }
};
