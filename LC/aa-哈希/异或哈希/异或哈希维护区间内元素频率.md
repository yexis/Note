class Solution {
public:
    using ull = uint64_t;
    static inline mt19937_64 rng = mt19937_64(chrono::steady_clock::now().time_since_epoch().count());

    vector<bool> validSubarrays(vector<int>& a, int K, vector<vector<int>>& qs) {
        int n = a.size(), m = qs.size();
        vector<ull> sum(n + 1);
        unordered_map<int, ull> hash;
        for (int i = 0; i < n; i++) {
            int x = a[i];
            if (!hash.count(x)) {
                hash[x] = rng();
            }
            sum[i + 1] = sum[i] ^ hash[x];
        }

        // cal:计算右端点为r时，区间内元素个数至少为k时的最大左端点
        // 那么如何计算右端点为r时，区间内元素个数 恰好 为k时的左端点范围？
        // 答案就是 [cal(k + 1), cal(k) - 1]
        auto cal = [&](int k) -> vector<int> {
            vector<int> L(n);
            unordered_map<int, int> cnt;
            int l = 0, r = 0;
            while (r < n) {
                cnt[a[r]]++;
                while (l <= r && cnt.size() >= k) {
                    if (--cnt[a[l]] == 0) cnt.erase(a[l]);
                    l++;
                }
                L[r] = l;
                r++;
            }
            return L;
        };

        auto L1 = cal(K + 1);
        auto L2 = cal(K);
        // 答案是则 [L1, L2 - 1]
        
        vector<bool> ans(m);
        for (int i = 0; i < qs.size(); i++) {
            auto& q = qs[i];
            int l = q[0], r = q[1];
            ans[i] = (sum[r + 1] == sum[l] && L1[r] <= l && l <= L2[r] - 1);
        }
        return ans;
    }
};
