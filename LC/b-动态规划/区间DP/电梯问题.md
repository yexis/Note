ll power(ll x, ll b, ll m = mod) {
    ll ans = 1;
    while (b) {
        if (b & 1) {
            ans *= x;
            ans %= m;
        }
        x *= x;
        x %= m;
        b >>= 1;
    }
    return ans;
}

class Solution {
public:
    long long elevatorRequests(int n, int start, vector<int>& a) {
        int m = a.size();
        sort(a.begin(), a.end());
        vector<int> b;
        for (int i = 0; i < m; i++) {
            if (a[i] != start) b.push_back(a[i]);
        } 
        a = b; m = a.size();
        if (m == 0) return 0;

        ll dp[m][m][2]; for (int i = 0; i < m; i++) for (int j = 0; j < m; j++) for (int k = 0; k < 2; k++) dp[i][j][k] = LLINF;

        for (int i = 0; i < m; i++) {
            dp[i][i][0] = dp[i][i][1] = 1ll * abs(a[i] - start) * m; 
        }

        for (int l = 2; l <= m; l++) {
            for (int i = 0; i + l - 1 < m; i++) {
                int j = i + l - 1;
                // 可来自于四个部分
                // dp[i + 1][j][0], dp[i + 1][j][1], dp[i][j - 1][0], dp[i][j - 1][1]

                // 当前的移动距离会对后续的所有位置产生影响，所以可以提前计算贡献
                // 这是阿贝尔定理？
                dp[i][j][0]
                dp[i][j][0] = min({
                    dp[i + 1][j][0] + 1ll * abs(a[i] - a[i + 1]) * (m - l + 1),
                    dp[i + 1][j][1] + 1ll * abs(a[i] - a[j]) * (m - l + 1),
                    dp[i][j - 1][0] + 2ll * abs(a[i] - a[j]) * (m - l + 1),
                    dp[i][j - 1][1] + 1ll * (abs(a[j] - a[j - 1]) + abs(a[j] - a[i])) * (m - l + 1),
                });

                // dp[i][j][1]
                dp[i][j][1] = min({
                   dp[i + 1][j][0] + 1ll * (abs(a[i] - a[i + 1]) + abs(a[j] - a[i])) * (m - l + 1),
                   dp[i + 1][j][1] + 2ll * abs(a[j] - a[i]) * (m - l + 1),
                   dp[i][j - 1][0] + 1ll * abs(a[j] - a[i]) * (m - l + 1),
                   dp[i][j - 1][1] + 1ll * abs(a[j] - a[j - 1]) * (m - l + 1), 
                });
            }
        }
        return min(dp[0][m - 1][0], dp[0][m - 1][1]);
    }
};
