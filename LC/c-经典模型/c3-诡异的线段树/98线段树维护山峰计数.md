using ll = long long;
struct Node {
    int z; // 懒加载
    
    int len;    // 峰值子数组的数量
    ll v;       // 区间长度
    int l, r;   // 左端点到第一个峰值的距离，右端点到最后一个峰值的距离；如果不存在则是-1
    
    Node() { 
        z = 0;
        v = 0;     
        len = 0;
        l = r = 0;
    }  
    Node(int _v, bool h) { 
        z = 0;
        v = 0; 
        len = 1;
        if (h) l = r = 1; 
        else l = r = -1;
    }
    // called by push_down
    void op() {

    }
};
// 自定义+
// ... 1 5 , 2 3...
Node operator+(const Node& a, const Node& b) {
    if (a.len == 0) return b;
    if (b.len == 0) return a;
    
    Node res;
    res.len = a.len + b.len;
    if (a.l != -1 && b.l != -1) {
        res.l = a.l;
        res.r = b.r;
        res.v = a.v + b.v + (1ll * a.len * b.len - 1ll * a.r * b.l);
    } else if (a.l != -1) {
        res.l = a.l;
        res.r = a.r + b.len;
        res.v = a.v + 1ll * (a.len - a.r) * b.len;
    } else if (b.l != -1) {
        res.l = a.len + b.l;
        res.r = b.r;
        res.v = 1ll * a.len * (b.len - b.l) + b.v;
    } else {
        res.l = res.r = -1;
        res.v = 0;
    }
    
    return res;
}

struct SegTree {
    int n;
    vector<int> a;
    Node tr[400010];
    SegTree(vector<int> _a) {
        a = _a;
        n = a.size();
    }
    bool judge(int i) {
        return (i > 0 && i < n - 1 && a[i] > a[i - 1] && a[i] > a[i + 1]);
    }
    void push_up(int o, int l, int r) {
        tr[o] = tr[o * 2] + tr[o * 2 + 1];
    }
    void push_down(int o, int l, int r) {
        if (tr[o].z) {
            tr[o * 2].op();
            tr[o * 2].z = tr[o].z;
            tr[o * 2 + 1].op();
            tr[o * 2 + 1].z = tr[o].z;
            tr[o].z = 0;
        }
    }
    void build(int o, int l, int r) {
        if (l == r) {
          	// init 单个元素
            bool peak = judge(l - 1);
            tr[o] = Node(a[l - 1], peak);
            return;
        }
        int m = (l + r) >> 1;
        build(o * 2, l, m);
        build(o * 2 + 1, m + 1, r);
        push_up(o, l, r);
    }
    void add(int o, int l, int r, int i, int u) {
        if (l == r) {
            bool peak = judge(l - 1);
            tr[o] = Node(a[l - 1], peak);
            return;
        }
        int m = (l + r) >> 1;
        if (i <= m) {
            add(o * 2, l, m, i, u);
        } else {
            add(o * 2 + 1, m + 1, r, i, u);
        }
        push_up(o, l, r);
    } 
    
    Node ask(int o, int l, int r, int L, int R) {
        if (L <= l && R >= r) {
            return tr[o];
        }
        push_down(o, l, r);
        int m = (l + r) >> 1;
        Node ans;
        if (L <= m) {
            ans = ans + ask(o * 2, l, m, L, R);
        }
        if (R > m) {
            ans = ans + ask(o * 2 + 1, m + 1, r, L, R);
        }
        return ans;
    }
};
class Solution {
public:
    vector<long long> countOfPeaks(vector<int>& a, vector<vector<int>>& qs) {
        int m = a.size(), n = qs.size();
        
        SegTree seg(a); seg.build(1, 1, m);
        
        vector<ll> res;
        for (int i = 0; i < n; i++) {
            auto& q = qs[i];
            if (q[0] == 1) {
                int l = q[1], r = q[2];
                Node ans = seg.ask(1, 1, m, l + 1, r + 1);
                res.push_back(ans.v);
            } else if (q[0] == 2) {
                int j = q[1], v = q[2];
                seg.a[j] = v;
                if (j - 1 >= 0) seg.add(1, 1, m, j, seg.a[j - 1]);
                seg.add(1, 1, m, j + 1, seg.a[j]);
                if (j + 1 < m) seg.add(1, 1, m, j + 2, seg.a[j + 1]);
            }
        }
        return res;
    }
};
