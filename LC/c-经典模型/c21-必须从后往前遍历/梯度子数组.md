ll power(ll x, ll b, ll m = mod) {
    ll ans = 1;
    while (b) {
        if (b & 1) {
            ans *= x;
            ans %= m;
        }
        x *= x;
        x %= m;
        b >>= 1;
    }
    return ans;
}

// 全局变量
constexpr static int N = 100000;
int first[N + 1];
deque<int> pos[N + 1];

class Solution {
public:
    // 50000 * 200
    int countStaircase(vector<int>& a) {
        int n = a.size();
        int ans = 0;

        for (int x : a) {
            first[x] = -1; pos[x].clear();
        }
        
        set<int> st; // 维护区间内的所有不同元素（按次序）
        
        // 逆序，方便从1开始枚举
        for (int i = n - 1; i >= 0; i--) {
            int x = a[i];
            if (first[x] != -1) st.erase(first[x]);
            st.insert(first[x] = i);
            
            pos[x].push_front(i);

            int R = n;
            auto it = st.begin();
            int good = 0, lst = i;
            for (int j = 1; ; j++) {
                // [i, nxt]
                int nxt = lst + j - 1; 
                if (nxt >= R) break; 
                // 按序枚举每个出现的元素
                while (it != st.end()) {
                    int x = a[*it];
                    // 此时判断第good + 1个是否满足，实际下标是good
                    int idx = good;
                    int m = (int)pos[x].size();
                    if (idx < m && pos[x][idx] <= nxt
                        && (idx + 1 >= m || pos[x][idx + 1] > nxt)
                    ) {
                        it++;
                        good++;
                        // 剪枝，收缩右边界
                        if (idx + 1 < m) R = min(R, pos[x][idx + 1]);
                        
                    } else {
                        break;
                    }
                }
                
                if (good == j) ans++;
                lst = nxt + 1;
            }
        }

        return ans;
    }
};
