##### 3 key points

###### 3.1 有向面积（鞋带公式）

将矩阵中的每个格子看作一个平面坐标点$x = col，y = row$，**即$x$轴向右，$y$轴向下**

对遍历到的所有点计算有向面积：$S = \sum_{i=0}^{len-1}(x_iy_{i+1} - x_{i+1}y_i)$，其中下标循环取模，即$p_{len}=p_0$

由于矩阵的$y$轴向下:

- $S > 0$：顺时针
- $S < 0$：逆时针

例如，一个矩形沿着上边向右、右边向下、下边向左、左边向上遍历，就是顺时针，此时 $S>0$。



如果是数学上的平面直角坐标系（**$x$轴向右，$y$轴向上**），则相反（待验证）

- $S < 0$：顺时针
- $S > 0$：逆时针



###### 3.2 转角和

笛卡尔坐标系（$x$轴向右，$y$轴向上）：

- $cross > 0$：位于向量左侧，即左转
- $cross < 0$：位于向量右侧，即右转
- $cross = 0$：直行或掉头



本题矩阵坐标系（$x$轴向右，$y$轴向下），则刚好相反：

- $cross > 0$：位于向量右侧，即右转
- $cross < 0$：位于向量左侧，即左转
- $cross = 0$：直行或掉头



利用叉积判断每次转弯的方向，把所有转向累加起来，就能判断整个多边形是顺时针还是逆时针；

**性质：不管多边形有多少个凸顶点、凹顶点，只要是简单闭合多边形，最终的有符号转弯次数之差始终是 4。**



随机选择一个方向遍历完所有点后，遍历过程重视$path$记录，依次遍历$path$中的每个节点，判断向量$(p[i] - p[i-1])$ 到 $(p[i + 1] - p[i])$ 的转角；**（最后一定是4次左转或者4次右转）**



###### 3.3 特殊点

选取最左上方的点$P$，考虑$P$在遍历过程$path$中的位置，找出点$P$的前驱节点和后继节点；

这两个点只可能是$P_{down}$和$P_{right}$，因为$P$是最左上方的点

根据$P$的前驱节点是$P_{down}$还是$P_{right}$，判断绕行方向是顺时针还是逆时针

- 如果前驱节点是$P_{down}$，则顺时针
- 如果前驱节点是$P_{right}$，则逆时针

 * 鞋带公式（有向面积）计算多边形绕向问题

int main() {
    ios;
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<string> g(n);
    for (auto &s : g) cin >> s;

    int sx, sy, tx, ty;
    cin >> sx >> sy >> tx >> ty;
    --sx; --sy;
    --tx; --ty;

    int dx[] = {-1, 0, 1, 0};
    int dy[] = {0, 1, 0, -1};

    auto get_neighbors = [&](int x, int y) {
        vector<pii> res;
        for (int d = 0; d < 4; d++) {
            int nx = x + dx[d];
            int ny = y + dy[d];
            if (nx >= 0 && nx < n &&
                ny >= 0 && ny < m &&
                g[nx][ny] == '*') {
                res.push_back({nx, ny});
            }
        }
        return res;
    };

    // 任选一个邻居，沿环遍历
    vector<pii> path;
    path.push_back({sx, sy});

    auto ns = get_neighbors(sx, sy);
    pii pre = {sx, sy};
    pii cur = ns[0];

    while (cur != pii{sx, sy}) {
        path.push_back(cur);

        auto v = get_neighbors(cur.first, cur.second);
        pii nxt = (v[0] == pre ? v[1] : v[0]);

        pre = cur;
        cur = nxt;
    }

    int len = path.size();

    // 找到终点在遍历序列中的位置
    int k = -1;
    for (int i = 0; i < len; i++) {
        if (path[i] == pii{tx, ty}) {
            k = i;
            break;
        }
    }

    // 鞋带公式：计算有向面积的两倍
    ll area = 0;

    // x轴向右，y轴向下
    for (int i = 0; i < len; i++) {
        auto [r1, c1] = path[i];
        auto [r2, c2] = path[(i + 1) % len];

        // 这里x对应c，y对应r
        area += 1LL * c1 * r2 - 1LL * c2 * r1;
    }

    // 矩阵坐标系下，area > 0 表示顺时针
    bool clockwise = area > 0;

    cout << (clockwise ? k : (len - k) % len) << '\n';

    return 0;
}

/*
 * https://ac.nowcoder.com/acm/contest/143199/D
 * 
 * 先随机找一个方向，遍历完所有的点，遍历过程记作path
 * 
 * 找到最左上方的点，设为P，考虑它的两个相邻点 P_down 和 P_right
 * 
 * 在遍历过程path中：
 * - 如果P的下一个节点是 P_right，那说明遍历方式是顺时针
 * - 如果P的下一个节点是 P_down，那说明遍历方式是逆时针
 * 
*/


int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<string> g(n);
    for (auto &s : g) cin >> s;

    int sx, sy, tx, ty;
    cin >> sx >> sy >> tx >> ty;
    --sx; --sy;
    --tx; --ty;

    int dx[] = {-1, 0, 1, 0};
    int dy[] = {0, 1, 0, -1};

    auto get_neighbors = [&](int x, int y) {
        vector<pii> res;
        for (int d = 0; d < 4; d++) {
            int nx = x + dx[d];
            int ny = y + dy[d];
            if (nx >= 0 && nx < n &&
                ny >= 0 && ny < m &&
                g[nx][ny] == '*') {
                res.push_back({nx, ny});
            }
        }
        return res;
    };

    vector<pii> path;
    path.push_back({sx, sy});

    auto ns = get_neighbors(sx, sy);
    pii pre = {sx, sy};
    pii cur = ns[0];

    while (cur != pii{sx, sy}) {
        path.push_back(cur);

        auto v = get_neighbors(cur.first, cur.second);
        pii nxt = (v[0] == pre ? v[1] : v[0]);

        pre = cur;
        cur = nxt;
    }

    int len = path.size();

    int k = -1;
    for (int i = 0; i < len; i++) {
        if (path[i] == pii{tx, ty}) {
            k = i;
            break;
        }
    }

    // 找到最上方、最靠左的点
    int pos = 0;

    for (int i = 1; i < len; i++) {
        if (path[i].first < path[pos].first ||
            (path[i].first == path[pos].first &&
             path[i].second < path[pos].second)) {
            pos = i;
        }
    }

    // 判断这个点的下一个点是否在右边
    pii nxt = path[(pos + 1) % len];

    bool clockwise = (nxt.second > path[pos].second);

    cout << (clockwise ? k : (len - k) % len) << '\n';

    return 0;
}
