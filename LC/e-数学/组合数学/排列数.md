ll power(ll x, ll b, ll m = mod) {
    ll ans = 1;
    while (b) {
        if (b & 1) {
            ans *= x;
            ans %= m;
        }
        x *= x;
        x %= m;
        b >>= 1;
    }
    return ans;
}

/*
 * 题意：构造谷点数为k的长度为n排列，求方案数
 * 求排列方案的基本套路：   从小到大枚举 或者 从大到小枚举 考虑插入位置
 * 本题的要求是计算股点数，很明显从大到小枚举要更好
 * 
 * 考虑插入位置
 * - 插入两侧，对谷点数没有影响
 * - 插入谷点两侧，对谷点没有影响
 * - 插入其他位置，谷点数 +1
 * 
 * 令 dp[i][j] 表示倒着枚举到i时，谷点数是j的方案数
 * dp[i][j] = 2 * dp[i + 1][j] + 2 * j * dp[i + 1][j] + (n - i + 1 - 2 - 2 * j) * dp[i + 1][j - 1]; 
 *
*/


void solve() {
    int n, K; cin >> n >> K;
    ll dp[n + 1][K + 1]; for (int i = 0; i <= n; i++) for (int j = 0; j <= K; j++) dp[i][j] = 0;
    
    dp[n][0] = 1;
    for (int i = n - 1; i >= 1; i--) {
        for (int j = 0; j <= K; j++) {
            mod_add(dp[i][j], 1ll * (2 + 2 * j) * dp[i + 1][j]);
            // n - i + 1 - 2 - 2 * (j - 1) = n - i - 2 * j + 1
            if (n - i - 2 * j + 1 > 0 && j - 1 >= 0) mod_add(dp[i][j], (n - i + 1 - 2 - 2 * (j - 1)) * dp[i + 1][j - 1]);
        }
    }
    cout << dp[1][K] << "\n";
}

int main() {

    while (T--) {
    	solve();
    }
