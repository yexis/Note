##### 2 examples

**元素$n$解成$k$个正整数的方案数**

> 隔板法，答案：$\binom{n-1}{k-1}$
>
> 证明：
>
> 将$n$分解成$n$个$1$, `1 1 1 1 1 ... 1 1 1`
>
> 那么，将$n$分解成$k$个正整数等价于 从$n$个$1$组成的$n - 1$个缝隙中选择$k-1$个进行插入



**元素$n$分解成$k$个奇数的方案数**

> 答案是 $\binom{\frac{n-k}{2}+k-1}{k-1}$
>
> 要求每个数是奇数，无法直接使用隔板法
> 每个数$a_i$是奇数，则令$a_i=2*b_i+1$，有$\sum_{i=1}^{k}(2*b_i+1) = n$
> $2*b_i = n - k$, 则$b_i = \frac{n-k}{2}$
>
> 
>
> **此时，$b_i$可以是任意正整数，不再受必须是奇数的条件限制，但是$b_i$可以是0，一个缝隙可以放置多个隔板**
>
> **所以无法使用常规隔板法，考虑最终态，一共有$\frac{n-k}{2}+k-1$个位置，选择$k-1$个放置隔板**
>
> 
>
> 如果，$n-k$是奇数，则无解，即不存在
> 否则，$n-k$是偶数，则使用隔板法，答案是 $\binom{\frac{n-k}{2}+k-1}{k-1}$

[4002. 统计有效序列数目](https://leetcode.cn/problems/count-valid-sequences/)

```cpp
// 1 1 | 1 1 1 | 1 | 1 1 1
    
// 将元素x，分解成k个正整数，有多少种方法
// 答案：使用隔板法，即在n-1个缝中插入k-1个隔板 C_{n-1}^{k-1}

// 将元素x，分解成k个奇数，有多少种方法
// 答案：
// 要求每个数是奇数，无法直接使用隔板法
// 每个数ai是奇数，则令ai=2*bi+1，有sum(2*bi+1) = n
// 2*bi = n - k, 则bi = (n - k) / 2
// 此时，bi可以是任意正整数，不再受必须是奇数的条件限制，但是bi可以是0
// n-k是奇数，则无解
// n-k是偶数，则使用隔板法，答案是 C_{(n-k)/2}^{k-1}
using ll = long long;
static constexpr int mod = 1e9 + 7;
ll qpow(ll n, ll b) {
    ll ans = 1;
    while (b) {
        if (b & 1) ans *= n, ans %= mod;
        n *= n, n %= mod;
        b >>= 1;
    }
    return ans;
}
struct Comb {
    ll fac[500010], inv[500010];
    Comb(int n) {
        fac[0] = inv[0] = 1;
        for (int i = 1; i <= n; i++) {
            fac[i] = fac[i - 1] * i; fac[i] %= mod;
            inv[i] = get_inv(fac[i]);
        }
    }
    ll get_inv(ll x) {
        return qpow(x, mod - 2);
    }
    ll C(int n, int m) {
        if (n < 0 || m < 0 || n < m) return -1;
        return fac[n] * inv[m] % mod * inv[n - m] % mod;
    }
};
class Solution {
public:
    // 序列中存在一个偶数，则乘积是偶数
    // 那么答案 = 所有方案数 - 序列全部是奇数的方案数
    int countValidSequences(int n, int K) {
        Comb comb(n);
        ll all = comb.C(n - 1, K - 1);
        if ((n - K) & 1) return all;
        
        int rst = (n - K) >> 1;
        ll odd = comb.C(rst + K - 1, K - 1);
        return (all - odd + mod) % mod;
    }
};
```
