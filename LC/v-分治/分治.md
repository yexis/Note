    ll ans = 1;
    while (b) {
        if (b & 1) {
            ans *= x;
            ans %= m;
        }
        x *= x;
        x %= m;
        b >>= 1;
    }
    return ans;
}

typedef int K;
K e() { return 0; }
int n;
K tr[100010];
void BIT(int nn) {
    n = nn;
    for (int i = 0; i <= n; i++) tr[i] = e();
}

int lowbit(int x) {
    return x & -x;
}

void add(int x, K u) {
    for (int i = x; i <= n; i += lowbit(i)) {
        tr[i] += u;
    }
} 

K ask(int x) {
    K ans = 0;
    if (x <= 0) return 0;
    for (int i = x; i > 0; i -= lowbit(i)) {
        ans += tr[i];
    }
    return ans;
}
K ask(int x, int y) {
    return ask(y) - ask(x - 1);
}


class Solution {
public:
    int m, n;
    vector<int> ob, a;
    int get(int x) {
        return lower_bound(ob.begin(), ob.end(), x) - ob.begin() + 1;

    ll solve(int l, int r) {
        if (l >= r) return 0;
        int mid = (l + r) >> 1;

        ll ans = 0;
        ans += solve(l, mid);
        ans += solve(mid + 1, r);

        // (a[i], b[i]), (a[j], c[j])
        int p1 = 0, p2 = 0;
        vector<pii> left(mid - l + 1), right(r - mid);
        multiset<int> ms;
        for (int i = mid; i >= l; i--) {
            auto it = ms.upper_bound(a[i]);
            int b, bi;
            if (it == ms.end()) b = INF;
            else b = *it;

            left[p1++] = pii(a[i], b);
            ms.insert(a[i]);
        }

        multiset<int> ms2;
        for (int i = mid + 1; i <= r; i++) {
            auto it = ms2.lower_bound(a[i]);
            int c, ci;
            if (it == ms2.begin()) c = -INF;
            else c = *prev(it);
            
            right[p2++] = pii(a[i], c);
            ms2.insert(a[i]);
        }

        sort(left.begin(), left.end(), [&](const auto& aa, const auto& bb) {
            return aa.second < bb.second;
        });

        sort(right.begin(), right.end(), [&](const auto& aa, const auto& bb) {
            return aa.first < bb.first;
        });

        BIT(m);
        
        // 双指针遍历
        // 从大到小遍历a[j]，将b[i]大于a[j]的i加入进BIT

        while (p >= 0) {

                q--;
            }
            int l = get(cj) - 1;
            int r = get(aj) - 1;
            ans += ask(r) - ask(l);
            p--;
        }

        return ans;
    };
    long long shadowPairs(vector<int>& a) {
        n = a.size();

        this -> a = a;
        this -> ob = a; 

        ob.push_back(-INF); ob.push_back(INF);
        sort(ob.begin(), ob.end());
        ob.erase(unique(ob.begin(), ob.end()), ob.end());
        m = ob.size();
        
        auto get = [&](int x) -> int {

        auto solve = [&](auto&& solve, int l, int r) -> ll{
            if (l >= r) return 0;
            int mid = (l + r) >> 1;
    
            ll ans = 0;
            ans += solve(solve, l, mid);

            // (a[i], b[i]), (a[j], c[j])

                else b = *it;

                ms.insert(a[i]);

                auto it = ms2.lower_bound(a[i]);

                else c = *prev(it);
                
                right[p2++] = pii(a[i], c);

                return aa.second < bb.second;

                }
                int l = get(cj) - 1;
                int r = get(aj) - 1;
                ans += ask(r) - ask(l);
