##### 3 key points

 **割边，一般是针对无向图**



对图深搜时，每一个节点只访问一次，被访问过的节点与边构成搜索树

时间戳`dfn[x]`：节点`x`第一次被访问的顺序

追溯值`low[x]`：从节点`x`出发，所能访问到的最早时间戳



**割边：**对于一个无向图，如果把一个边删除后，连通块的个数增加了，那么这个点就是割边，又称为桥

**割点判定法则：**

- 当搜搜树上存在`x`的一个子节点`y`，满足`low[y] > dfn[x]`，则`(x,y)`这条边就是割边



`low[y] > dfn[x]`，说明从`y`出发，在不经过`(x,y)`这条边的前提下，不管走哪条边，都无法到达`x`或比`x`更早访问的节点。

故删除点`(x,y)`这条边后，以`y`为根的子树subtree(y)也就断开了，即环外的边割得段。

反之，若`low[y] <= dfn[x]`，则说明`y`能绕行其他边到达`x`或比`x`更早访问的节点，`(x，y)`就不是割边了，即环内的边割不掉。

/*
 * 割边
 * 无向图

 * 
 * P3388
 * O(n + m)

struct CutEdge {
    int n, m, tot, root;

    // (u, v, id)
    int edge_id;
    vector<vector<pii>> g;

    // 时间戳，追溯值
    vector<int> dfn, low;
    
    // 是否是割点
    vector<int> cut;

    // 是否是割边
    vector<int> cut_edge;

    CutEdge(int nn, int mm) {
        n = nn; m = mm; tot = 0; edge_id = 0;
        g.resize(n); dfn.resize(n, -1); low.resize(n, -1);
        cut.resize(n), cut_edge.resize(m);
    }

    void tarjan(int x, int last_id) {
        dfn[x] = low[x] = tot++;
        int child = 0; // 符合条件的子树个数
        for (auto [y, curr_id] : g[x]) {
            // 跳过返祖边
            if (curr_id == last_id) continue;

            if (dfn[y] == -1) { // 若y尚未访问
                tarjan(y, curr_id);
                // 回x时，更新low，判割点
                // 注意这里取的是low[y]
                low[x] = min(low[x], low[y]);
                if (low[y] >= dfn[x]) {
                    child++;
                    if (x != root || child > 1) {
                        cut[x] = true;
                    }
                }
                // 判割边
                if (low[y] > dfn[x]) {
                    cut_edge[curr_id] = true;
                }
            } else { // 若y已经访问
                // 注意这里取的是dfn[y]
                low[x] = min(low[x], dfn[y]);
            }
        }
    }
    void add_edge(int u, int v) {
        g[u].push_back(pii(v, edge_id));
        g[v].push_back(pii(u, edge_id));
        edge_id++;
    }
    void set_root(int rt) {
        root = rt;
    }
};

void solve() {
    int n, m; cin >> n >> m;
    
    CutEdge ce(n, m);
    for (int i = 0; i < m; i++) {
        int u, v; cin >> u >> v; u--, v--;
        ce.add_edge(u, v);
    }

    for (int i = 0; i < n; i++) {
        if (ce.dfn[i] == -1) {
            ce.set_root(i);
            ce.tarjan(i, -1);
        }
    }
    
    int cnt = 0;
    for (int i = 0; i < m; i++) {
        if (ce.cut_edge[i]) {
            cnt++;
        }
    }
    cout << cnt << "\n";
}

    	solve();
    }

}
