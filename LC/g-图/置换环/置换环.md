置换环的遍历方式

```cpp
    int id = 0; // 环编号
    map<int, int> size; // 环大小
    vector<int> loop(n + 1, -1); // 点i属于哪个环
    for (int i = 1; i <= n; i++) {
        if (a[i] == i) {
            continue;
        }
        if (loop[i] != -1) {
            continue;
        }
        int now = ++id, sz = 0;
        loop[i] = now; sz++;
        for (int x = a[i]; x != i; x = a[x]) {
            loop[x] = now; sz++;
        }
        size[now] = sz;
    }
		
		// 统计需要置换的次数
    int cnt = 0; for (auto& [id, sz] : size) cnt += (sz - 1);
```
