##### 3 key points

**差分约束系统是给$m$个$n$元一次不等式，不等式形如$x_i - x_j \le c_k$，其中$c_k$是常数（0、正数或负数）**

求一组解$\lbrace x_1,x_2,...,x_n \rbrace = \lbrace a_1, a_2, ..., a_n \rbrace$，使得所有不等式（约束条件）得到满足，否则判断无解



不等式 $x_i - x_j \le c_k$ 可以转化成 $x_i \le x_j + c_k$，这类似单源最短路的三角形不等式（松弛操作）$d_v <= d_u + w$

**因此，把每个变量$x$看作一个点，每个不等式 $x_i - x_j \le c_k$ ，则从点$x_j$向点$x_i$连一条长度为$c_k$的有向边**

- 跑 SPFA 先把所有点均入队，避免走不到环. 或者从超级源点 0 向各点连 0 边权的边，从 0 点开始跑

- 可行解是点之间的相对距离，d 数组初始值均取 0 即可



 差分约束系统求解不等式组有两种方法：最短路和最长路

- **$x_a <= x_b + c$，$c$是常数，跑最短路，判负环，求最大解**

- **$x_a >= x_b + c$，$c$是常数，跑最长路，判正环，求最小解**

在固定参考变量、系统可行且相应极值有界的前提下：

- 最短路求的是各变量相对于参考变量的最大可行差值，即最大解。

- 最长路求的是各变量相对于参考变量的最小可行差值，即最小解。

##### 4 thought

**适用场景：**

- **给定一组不等式，判断约束是否满足，即计算不等式是否 存在/不存在 可行解**

  > **存在可行解：有向图不存在 正/负 环**（基于基准点，找点$x$最大的解或者最小的解）
  >
  > **不存在可行解：有向图不存在 正/负 环**（一般找最值使得约束不满足）

ll power(ll x, ll b) {
    ll ans = 1;
    while (b) {
        if (b & 1) {
            ans *= x;
            ans %= mod;
        }
        x *= x;
        x %= mod;
        b >>= 1;
    }
    return ans;
}

/*
 * https://www.luogu.com.cn/problem/P4926
 * 
 * 最短路：spfa算法，基于松弛操作的单源最短路算法，Bellman-Ford算法的优化版本
 * 每轮只访问上一轮更新过的点，因为只有上一轮被更新过的点，这一轮才有可能被更新
 * 使用队列进行维护；
 * vis[u]记录u点是否在队内；
 * cnt[v]记录边数，判负权环
 * 
 * 
 * 差分约束系统求解不等式组有两种方法：最短路和最长路
 * - a - b <= c，跑最短路，判负环
 * - a - b >= c，跑最长路，判正环
 * 
 * 在固定参考变量、系统可行且相应极值有界的前提下：
 * - 最短路求的是各变量相对于参考变量的最大可行差值。
 * - 最长路求的是各变量相对于参考变量的最小可行差值。
 * 
 * 时间复杂度：O(n * m) 
 * n：节点数，m：边数
*/

struct DifferenceConstraints {
    // 点数，边数
    int n, m;
    vector<pii> edges;
    vector<vector<pii>> g;
    vector<int> d, cnt, vis, pre;
    DifferenceConstraints(int nn, int mm) {
        n = nn, m = mm; 
        g.resize(n); init();
    }
    void init() {
        d.assign(n, INF); cnt.assign(n, 0);
        vis.assign(n, 0); pre.assign(n, 0);
    }
    void add_edge(int u, int v, int w) {
        edges.push_back(pii(u, v));
        g[u].push_back(pii(v, w));
    }
    // a - b <= c，最短路，求最大解
    // 返回true，表示有负环， false表示无负环
    // s >= n 全源判负环; s < n 无负环，最短路找最大解
    bool spfa(int s) {
        int n = g.size();
        
        queue<int> q;
        if (s >= n) {
            for (int i = 0; i < n; i++) { 
                q.push(i); vis[i] = true; d[i] = 0; 
            }
        } else { 
            q.push(s); vis[s] = true; d[s] = 0;
        }
        
        while (q.size()) {
            auto u = q.front(); q.pop(); vis[u] = false;
            for (auto& [v, w] : g[u]) {
                if (d[u] + w < d[v]) {
                    d[v] = d[u] + w; cnt[v] = cnt[u] + 1;
                    pre[v] = u;
                    if (cnt[v] >= n) return true;
                    if (!vis[v]) {
                        q.push(v); vis[v] = true;
                    }
                }
            }
        }
        return false;
    }

    // a - b >= c，最长路，求最小解
    // 返回true，表示有正环
    // 返回false，表示无正环
    bool spfa2(int s) {
        int n = g.size();
        
        queue<int> q;
        if (s >= n) {
            for (int i = 0; i < n; i++) { 
                q.push(i); vis[i] = true; d[i] = 0; 
            }
        } else { 
            q.push(s); vis[s] = true; d[s] = 0;
        }
        
        while (q.size()) {
            auto u = q.front(); q.pop(); vis[u] = false;
            for (auto& [v, w] : g[u]) {
                if (d[u] + w > d[v]) {
                    d[v] = d[u] + w; cnt[v] = cnt[u] + 1;
                    pre[v] = u;
                    if (cnt[v] >= n) return true;
                    if (!vis[v]) {
                        q.push(v); vis[v] = true;
                    }
                }
            }
        }
        return false;
    }
 };

 /*
  A[i]: i时刻需要的人数
  B[i]: i时刻到来的人数
  C[i]: i时刻雇佣的人数

  条件1：i时刻雇佣的人数需要 小于等于 到来的人数，即 C[i] <= B[i]
  条件2：从i-7到i时刻雇佣的人都能在都能在i时刻工作，即 x[i - 6] + ... + x[i + 1] >= A[i]

  求最小解，跑最长路，约束全部转化成 >=
  1. s[i + 1] - s[i] >= 0           => s[i + 1] >= s[i] + 0
  2. s[i] - s[i + 1] >= -B[i]       => s[i] >= s[i + 1] - B[i]
  3. s[i + 1] - s[i - 7] >= A[i], 7 <= i <= 23
    => s[i + 1] >= s[i - 7] + A[i]
  4. s[i + 1] + (s[24] - s[24 - (8 - (i + 1)]) >= A[i], i < 7
    => s[i + 1] - s[16 + i + 1] >= A[i] - s[24]
    => s[i + 1] >= s[16 + i + 1] >= A[i] - s[24]

  s[24]即题目所求

  二分mid，令 mid = s[24]，则有 mid <= s[24] <= mid
  5. s[24] >= mid
    => s[24] >= s[0] + mid
  6. -s[24] >= -mid
    => -s[24] >= -s[0] - mid
    => s[0] >= s[24] - mid

  答案：
  当 mid > ans 时，不等式满足，说明有解，无正环
  当 mid <= ans 时，不等式不满足，说明无解，有正环 
*/
void solve() {
    // A: 时刻i需要的人数
    vector<int> A(24, 0); for (int i = 0; i < 24; i++) cin >> A[i];
    
    int n; cin >> n;
    // B: 时刻i应聘的人数
    vector<int> B(24, 0);  
    for (int i = 0; i < n; i++) {
        int e; cin >> e;
        B[e]++;
    }

    auto check = [&](int mid) -> bool {
        DifferenceConstraints dc(25, 1000);
        for (int i = 0; i < 24; i++) {
            // cerr << i << " " << i + 1 << "\n";
            dc.add_edge(i, i + 1, 0);
            dc.add_edge(i + 1, i, -B[i]);
        }
        for (int i = 0; i < 7; i++) {
            dc.add_edge(i + 17, i + 1, A[i] - mid);
        }
        for (int i = 7; i < 24; i++) {
            dc.add_edge(i - 7, i + 1, A[i]);
        }
        dc.add_edge(0, 24, mid);
        dc.add_edge(24, 0, -mid);

        bool has = dc.spfa2(25);
        if (has) return false;
        return true;
    };

    // C: 时刻i雇佣的人数
    int l = 0, r = n, ans = -1;
    while (l <= r) {
        int mid = (l + r) >> 1;
        if (check(mid)) {
            ans = mid;
            r = mid - 1;
        } else {
            l = mid + 1;
        }
    }
    if (ans == -1) {
        cout << "No Solution" << "\n";
        return;
    }
    cout << ans << "\n";
}



int main() {
    int T = 1;
    cin >> T;
    while (T--) {
        solve();    
    }
    return 0;
}
