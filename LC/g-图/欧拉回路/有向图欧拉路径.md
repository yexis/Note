##### 3 key points

**欧拉路径：**每条边恰好经过一次的路径

**欧拉回路：**每条边恰好经过一次又回到起点的路径



**有向图欧拉路径的判定：**有且仅有两个点的入度不等于出度，出度与入度多1的点为起点，入度比出度多1的点为终点

**有向图欧拉回路的判定：**所有点的入度等于出度



适用场景：

- 每个字符串使用一次
- 字典序最小
- 每条边恰好经过一次
- 每条边恰好从两个方向经过一次

ll power(ll x, ll b, ll m = mod) {
    ll ans = 1;
    while (b) {
        if (b & 1) {
            ans *= x;
            ans %= m;
        }
        x *= x;
        x %= m;
        b >>= 1;
    }
    return ans;
}

struct EulerPath {
    int n, m;
    int edge_id;
    vector<vector<pii>> g;
    vector<int> deg_in, deg_out;

    vector<vector<int>> edges;
    // 边是否被删除
    vector<int> ptr;

    // 栈，后序输出
    stack<int> stk;

    EulerPath(int nn, int mm) {
        n = nn, m = mm;
        edge_id = 0; g.resize(n); 
        deg_in.resize(n); deg_out.resize(n);
        ptr.resize(n);
    }
    void add_edge(int u, int v) {
        edges.push_back({u, v});
        g[u].push_back(pii(v, edge_id++));
        deg_out[u]++, deg_in[v]++;
    }
    void sort_edges() {
        for (auto& es : g) {
            sort(es.begin(), es.end());
        }
    }
    // 递归
    void dfs(int u) {
        // 这里不能写成 for (auto [v, _] : g[u]) 
        // 因为会存在环 1->2->1->....->1->2->1
        for (int i = ptr[u]; i < g[u].size(); i = ptr[u]) {
            ptr[u] = i + 1;
            dfs(g[u][i].first);
        }
        stk.push(u);
    }
    // 递推
    void search(int u) {
        stack<int> q; q.push(u);
        while (q.size()) {
            int x = q.top();
            if (ptr[x] == g[x].size()) {
                stk.push(x);
                q.pop();
            } else {
                int y = g[x][ptr[x]].first;
                q.push(y);
                ptr[x]++;
            }
        }
    }
    // 最后完成读栈后，最好判断一下边数是否等于m
    bool is_valid(int& start, int& end) {
        int bad = 0;
        start = 0, end = 0;
        for (int i = 0; i < n; i++) {
            if (deg_in[i] == deg_out[i]) continue;
            bad++;
            if (deg_in[i] + 1 == deg_out[i]) start = i;
            else if (deg_in[i] - 1 == deg_out[i]) end = i;
            else return false;
        } 
        if (bad == 1 || bad > 2) return false;
        // 注意：bad只能是0（欧拉回路）或者2（欧拉路径）
        if (bad == 0) {
            // 注意：start是否会越界
            start = 0; while (!deg_out[start]) start++; end = start;
        } else if (start == -1 || end == -1) {
            return false;
        }
        return true;
    }
};

void solve() {
    int m; cin >> m;
    vector<string> words(m); for (string& w : words) cin >> w;
    sort(words.begin(), words.end(), [&](const auto& aa, const auto& bb) {
        return aa + "." < bb + ".";
    });

    vector<string> W[26][26];
    EulerPath el(26, m);
    for (int i = 0; i < m; i++) {
        int b0 = words[i][0] - 'a', e0 = words[i].back() - 'a';
        el.add_edge(b0, e0);
        W[b0][e0].push_back(words[i]);
    }

    int start = -1, end = -1;
    bool valid = el.is_valid(start, end);
    if (!valid) {
        cout << "***" << "\n";
        return;
    }

    el.dfs(start);

    string ans;

    int ptr[26][26]; 
    memset(ptr, 0, sizeof(ptr));
    int pre = -1, es = 0;
    while (el.stk.size()) {
        if (pre != -1) {
            int curr = el.stk.top();
            int& pt = ptr[pre][curr];
            ans += W[pre][curr][pt] + ".";
            es++;
            pt++;
        }
        pre = el.stk.top();
        el.stk.pop();
    }

    if (es != m) {
        cout << "***" << "\n";
        return;
    }
    
    ans = ans.substr(0, ans.size() - 1);
    cout << ans  << "\n";
}
