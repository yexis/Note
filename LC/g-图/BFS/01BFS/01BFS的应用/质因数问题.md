###### 4.1 普通BFS（错误解）

$10^{14}$以内的数，最多具有$17280$个因子，按照每个余数$r$作为节点，$N$的因子作为边，那么将是一个 $M(2*10^5)$个节点，$E(17280)$条边的有向图，每条边的权值是$1$；若对于这个图进行普通BFS，复杂度可以达到$O(M * E) $

为什么是$O(M * E)$，考虑每个节点的出边，每次遍历完后，需要遍历该节点的所有出边



###### 4.2 最短路 01-BFS

将$N$写成质因数形式，即 $p_0^{a_0}p_1^{a_1}p_2^{a_2},...,p_k^{a_k}$，其中$\sum_{i=0}^{k}a_i \le 64$

将$(j, r)$定义成当前需要考虑第$j$个质因数且当前余数为$r$的状态，那么考虑所有的状态$(j,r)$即可求得答案

> 为什么？
>
> 思考一下，不难发现 状态$(j,r)$ 就是最短路建模中的 唯一态；那么对于状态$(j,r)$跑最短路，那么状态总数是$O(M*log(N))$
>
> 时间复杂度为什么是 $O(M * long(n))$呢？
>
> 01-BFS的时间复杂度是 （状态数 * 转移次数）；来看一下本题，状态数一共有$M * D$个，而对于每个状态$(j,r)$，其转移次数是$a_j$个；
>
> 而 $(a_j+1)*D ≈ log(N)$，所以整体的时间复杂度是$(O(M * D* a_j))$，即 $O(M * log(N)$

    ll ans = 1;

        x %= m;
        b >>= 1;

void solve() {
    ll N;
    int M;
    cin >> N >> M;

    // N = 1 的特殊情况
    if (N == 1) {
        for (int k = 0; k < M; k++) {
            cout << (k == 1 % M ? 1 : -1) << '\n';
        }
        return;
    }

    // 质因数分解
    vector<pair<ll, int>> fac;

    for (ll p = 2; p * p <= N; p++) {
        if (N % p) continue;

        int cnt = 0;
        while (N % p == 0) {
            N /= p;
            cnt++;
        }

        fac.push_back({p % M, cnt});
    }

    if (N > 1) {
        fac.push_back({N % M, 1});
    }

    int D = fac.size();

    // dist[i][r]：状态 (i,r) 的最短距离
    vector<vector<int>> dist(D, vector<int>(M, INF));

    deque<tuple<int, int, int>> q;

    dist[0][1 % M] = 0;
    q.push_front({0, 1 % M, 0});

    // 01-BFS
    while (!q.empty()) {
        auto [i, r, d] = q.front();
        q.pop_front();

        if (d != dist[i][r]) continue;

        int ni = (i + 1) % D;
        int w = (i == 0 ? 1 : 0);
        int nd = d + w;

        ll cur = r;

        // 枚举当前质因子的指数
        for (int e = 0; e <= fac[i].second; e++) {

            int nr = cur;

            if (nd < dist[ni][nr]) {
                dist[ni][nr] = nd;

                if (w == 0) {
                    q.push_front({ni, nr, nd});
                } else {
                    q.push_back({ni, nr, nd});
                }
            }

            cur = cur * fac[i].first % M;
        }
    }

    // 统计答案
    for (int r = 0; r < M; r++) {
        int ans = INF;

        for (int i = 0; i < D; i++) {
            ans = min(ans, max(1, dist[i][r]));
        }

        cout << (ans == INF ? -1 : ans) << '\n';
    }
}

    	solve();
    }
    return 0;
}
