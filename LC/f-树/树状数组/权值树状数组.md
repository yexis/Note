##### 3 key points

- 权值树状数组求第k小的元素 $O(n \times long(n))$
- 权值树状数组求第大小的元素 $O(n \times long(n))$，即 求第 $n - k + 1$小的元素
- 权值树状数组求前k小的元素和 $O(n \times long(n))$
- 权值树状数组求前k大的元素和 $O(n \times long(n))$，即 求前 $sum$ - 前 $n - k$ 小的元素和

using ll = long long;
struct BIT {
    vector<int> a;
    vector<int> cnt;
    vector<ll> sum;
    int n, m;
    BIT(vector<int>& aa) {
        a = aa;
        n = aa.size();
        m = (31 - __builtin_clz(n));
        cnt.resize(n + 1);
        sum.resize(n + 1);
    }
    int lowbit(int x) {
        return x & -x;
    }
    // 第x个位置增加u个val
    // x是val的权值
    void add(int x, int u, int val) {
        for (int i = x; i <= n; i += lowbit(i)) {
            cnt[i] += u;
            sum[i] += u * val;
        }
    }
    int ask(int x) {
        int ans = 0;
        for (int i = x; i > 0; i -= lowbit(i)) {
            ans += cnt[i];
        }
        return ans;
    }

    // 权值树状数组
    // 查询第 k 小（k从1开始）
    int kth(int k) {
        int i = 0;
        for (int b = (1 << m); b; b >>= 1) {
            int nxt = i | b;
            if (nxt < n && cnt[nxt] < k) {
                k -= cnt[nxt];
                i = nxt;
            }
        }
        return a[i];
    }

    // 权值树状数组
    // 前k小的元素和（k从1开始）
    // 查询前k大的元素和，就是查询前n - k小的元素和 （k从1开始）
    ll pre_sum(int k) {
        ll s = 0;
        int i = 0;
        for (int b = (1 << m); b; b >>= 1) {
            int nxt = i | b;
            if (nxt < cnt.size() && cnt[nxt] < k) {
                k -= cnt[nxt];
                s += sum[nxt];
                i = nxt;
            }
        }
        // 加上等于第k小的数a[i]
        // 此时k是a[i]剩余的次数
        return s + 1ll * a[i] * k;
    }
};

class Solution {
public:
    using ll = long long;
    const ll INF = 1e17;
    long long maxSum(vector<int>& a, int k) {
        int n = a.size();

        vector<int> ob = a; // 离散化数组
        sort(ob.begin(), ob.end());
        ob.erase(unique(ob.begin(), ob.end()), ob.end());
        vector<int> rank(n); // rank[i]: a[i]离散化后的值

        BIT all_tree(ob);
        ll total = 0;
        for (int i = 0; i < n; i++) {
            int x = a[i];
            rank[i] = lower_bound(ob.begin(), ob.end(), x) - ob.begin() + 1;
            all_tree.add(rank[i], 1, x);
            total += x;
        }


        ll ans = LLONG_MIN;
        for (int left = 0; left < n; left++) {
            BIT in_tree(ob);
            BIT out_tree = all_tree;
            int need_swap = 0;
            ll sub_sum = 0;
            for (int right = left; right < n; right++) {
                int x = a[right];
                int rk = rank[right];

                // 子数组内的元素和
                sub_sum += x;

                in_tree.add(rk, 1, x);
                out_tree.add(rk, -1, x);

                bool inc = false;
                int sz = right - left + 1;

                // 是否可以增加交换次数
                if (need_swap < k && need_swap < sz && need_swap < n - sz) {
                    // 内部的第k小
                    // 外部的第(n - sz - k + 1)大
                    if (in_tree.kth(need_swap + 1) < out_tree.kth(n - sz - (need_swap + 1) + 1)) {
                        inc = true;
                        need_swap++;
                    }
                } 
                // 是否可以减少交换次数
                if (!inc && need_swap > 0) {
                    if (in_tree.kth(need_swap) >= out_tree.kth(n - sz - need_swap + 1)) {
                        need_swap--;
                    }
                }

                ll delta = 0;
                if (need_swap > 0) {
                    ll in_sum = in_tree.pre_sum(need_swap);
                    ll out_sum = total - sub_sum - out_tree.pre_sum(n - sz - need_swap);
                    delta = out_sum - in_sum;
                }

                ans = max(ans, sub_sum + delta);
            }
        }

        return ans;
    }
};
