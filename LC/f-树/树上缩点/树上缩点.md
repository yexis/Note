    ll ans = 1;
    while (b) {

        x *= x;
        x %= m;
        b >>= 1;

struct DJ {
    vector<int> fa;
    vector<int> ra;
    vector<int> cnt; // 每个连通分量的大小
  	int tot; // 联通块的总数
    int n;
    DJ(int _n) {
        n = _n;
        tot = _n;
        fa.resize(n);
        ra.resize(n);
        cnt.resize(n, 1);
        iota(fa.begin(), fa.end(), 0);
    }

    int find(int x) {
        if (x != fa[x]) {
            fa[x] = find(fa[x]);
        }
        return fa[x];
    }

    void merge(int x, int y) {
        int rx = find(x);
        int ry = find(y);
        if (rx == ry) {
            return;
        }
        if (ra[rx] < ra[ry]) {
            swap(rx, ry);
        }
        tot--;
        fa[ry] = rx;
        cnt[rx] += cnt[ry];
        if (ra[rx] == ra[ry]) {
            ra[rx]++;
        }
    }

  	bool conn(int x, int y) {
    	return find(x) == find(y);
    }
};

struct MergeNodeTree {
    // rt: 新图的起点
    int n, rt;
    vector<pii> edges;
    // g:原树 ng:新树
    vector<vector<pii>> g, ng;
    vector<int> colors; // 点的颜色，相邻相同颜色的点合并
    DJ* dj;
    MergeNodeTree(int nn) {
        n = nn, rt = 0;
        g.resize(n); ng.resize(n); 
        dj = new DJ(n);
    }
    void add_edge(int u, int v, int w) {
        edges.push_back(pii(u, v));
        g[u].push_back(pii(v, w));
        g[v].push_back(pii(u, w));
    }
    void merge(vector<int>& cls) {
        colors = cls;
        
        for (auto [u, v] : edges) {
            if (colors[u] == colors[v]) dj->merge(u, v);
        }
        
        for (auto [u, v] : edges) {
            int ru = dj->find(u), rv = dj->find(v);
            rt = ru;
            if (ru == rv) continue;
            pii pr = pii(ru, rv);
            ng[ru].push_back(pii(rv, -1));
            ng[rv].push_back(pii(ru, -1));
        }
    }
};

void solve() {
    int n; cin >> n;
    vector<int> C(n); for (int i = 0; i < n; i++) cin >> C[i];
    
    MergeNodeTree mnt(n);
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v; u--, v--;
        mnt.add_edge(u, v, -1);
    }
    mnt.merge(C);

    auto& dj = mnt.dj;

    ll ans = 0; 
    auto dfs = [&](auto&& dfs, int u, int o, int co) -> void {
        unordered_map<int, int> mp;
        for (auto& [v, _] : mnt.ng[u]) if (v != o) {
            ll curr_c = dj->cnt[dj->find(v)];
            if (o != -1 && C[o] == C[v]) {
                ans += 1ll * dj->cnt[o] * curr_c;
            }
            ans += 1ll * mp[C[v]] * curr_c;
            mp[C[v]] += curr_c;
            dfs(dfs, v, u, C[u]);
        }
    };
    dfs(dfs, mnt.rt, -1, -1);
    cout << ans << "\n";
}

    // cin >> T;
    while (T--) {
    	solve();
    }

}
