###### 树上差分

- 修改子树$u$：转化为DFS序后的差分，修改$[L[u], R[u]]$

- 修改$u$的子树$v$：转化为DFS序后的差分，修改 $[L[v], R[v]]$

- 修改$u$除$v$外的子树：转化为DFS序后的差分，修改两部分 $[L[u], L[v] - 1]$ 和 $[R[v] + 1, R[u]]$

        b >>= 1;

// ********************** 模板开始 **********************
using ll = long long;
using pii = pair<int, int>;
struct TREE {
    int n, m, tot, max_dep;
    vector<vector<int>> g;
    // L: 进入dfs序， 从1开始
    // R: 离开dfs序
    vector<int> L, R;
    // L[u]->u, R[u]->u 映射
    vector<int> reverseL, reverseR;
    // dep: 深度从0开始
    // sum: 观察权重位于点上还是边上?
    vector<int> dep, sum;  
    // P: 记录每一层上的节点(按DFS序)
    vector<vector<int>> P;
    // V: 若要基于第k层的孩子区间询问，可转RMQ
    vector<vector<int>> V;
    // 倍增数组 st
    vector<vector<int>> st;

    TREE(int nn, vector<vector<int>>& gg) {
        g = gg; 
        max_dep = 0;
        n = nn; m = 32 - __builtin_clz(n); tot = 0;
        L.resize(n); R.resize(n);
        reverseL.resize(n + 1); reverseR.resize(n + 1);
        P.resize(n), V.resize(n);
        dep.resize(n), sum.resize(n);
        // 倍增数组
        st.resize(n, vector<int>(m, -1));

        auto dfs = [&](auto&& dfs, int u, int o) -> void {
            L[u] = ++tot; reverseL[tot] = u;
            max_dep = max(max_dep, dep[u]);
            P[dep[u]].push_back(L[u]);
            st[u][0] = o;
            for (auto& v : g[u]) {
                if (v == o) continue;
                dep[v] = dep[u] + 1;
                // sum[v] = sum[u] + ?
                dfs(dfs, v, u);
            }
            R[u] = tot; 
        };
        dfs(dfs, 0, -1);

        // 计算倍增数组
        for (int i = 1; i < m; i++) {
            for (int u = 0; u < n; u++) {
                if (st[u][i - 1] != -1) {
                    st[u][i] = st[st[u][i - 1]][i - 1];
                }
            }
        }
    }

    // log(c)查询深度差为c的祖先节点
    // 若u是v的祖先节点，可求u的孩子节点中，也是v的祖先节点的那一个
    int get_ancestor(int u, int c) {
        for (int j = 0; j < m; j++) {
            if ((c >> j) & 1) u = st[u][j];
        }
        return u;
    }

    // 2. 根据倍增数组计算lca
    int get_lca(int x, int y) {
        // 将 y 置于 较深的位置
        if (dep[x] > dep[y]) {
            swap(x, y);
        }

        // 让 x 和 y 位于同一深度
        int d = dep[y] - dep[x];
        for (; d; d &= d - 1) {
            int i = __builtin_ctz(d);
            y = st[y][i];
        }
        if (y == x) {
            return x;
        }

        // 从大到小往上跳
        for (int i = m - 1; i >= 0; i--) {
            if (st[x][i] != st[y][i]) {
                x = st[x][i];
                y = st[y][i];
            }
        }

        // 最后返回x的父节点
        return st[x][0];
    }

    // 计算点x到点y的距离
    int get_dist(int x, int y) {
        int ca = get_lca(x, y);
        return dep[x] + dep[y] - 2 * dep[ca];
    }

    // 获取节点u第k层的所有孩子
    // 返回所有节点的 进入dfs序 (即L[u])
    // 若为区间询问，转RMQ问题
    void get_k_child(int u, int k, vector<int>& res) {
        if (dep[u] + k > max_dep) return;
        auto& p = P[dep[u] + k];
        int idx1 = lower_bound(p.begin(), p.end(), L[u]) - p.begin();
        int idx2 = upper_bound(p.begin(), p.end(), R[u]) - p.begin() - 1;
        for (int i = idx1; i <= idx2; i++) {
            res.push_back(reverseL[p[i]]);
        }
        return;
    }
};
// ********************** 模板结束 **********************

void solve() {
    int n, K; cin >> n >> K;
    vector<vector<int>> g(n);
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v; u--, v--;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    TREE lca(n, g);

    vector<int> Diff(n + 3, 0);
    while (K--) {
        int op, u, v; cin >> op >> u >> v; u--, v--;
        int lc = lca.get_lca(u, v);
        int dp1 = lca.dep[u], dp2 = lca.dep[v];
        if (op == 1) {
            // u是v的祖先
            if (lc == u) {
                // u的子树方向全删 v
                int delta = dp2 - dp1 - 1;
                int uv = lca.get_ancestor(v, delta);
                int l = lca.L[uv], r = lca.R[uv]; 
                Diff[l]--, Diff[r + 1]++;
                
            } else {
                int l = lca.L[u], r = lca.R[u];
                // [1, l - 1] 全删
                if (1 <= l - 1) Diff[1]--, Diff[l]++;
                // [r + 1, n] 全删
                if (r + 1 <= n) Diff[r + 1]--, Diff[n + 1]++;
            }
        } else if (op == 2) {
            // u和v没有祖先关系
            if (lc == u || lc == v) {
                if (lc == v) swap(u, v);
                // u的父亲方向
                int l = lca.L[u], r = lca.R[u];
                // [1, l - 1] 全删
                if (1 <= l - 1) Diff[1]--, Diff[l]++;
                // [r + 1, n] 全删
                if (r + 1 <= n) Diff[r + 1]--, Diff[n + 1]++;
                
                // 删除u
                Diff[l]--, Diff[l + 1]++;

                // u的子树方向 除v外
                int delta = dp2 - dp1 - 1;
                int uv = lca.get_ancestor(v, delta);
                int lu = lca.L[u], ru = lca.R[u];
                l = lca.L[uv], r = lca.R[uv];  // 不删
                // [lu + 1, l - 1] 全删
                if (lu + 1 < l - 1) Diff[lu + 1]--, Diff[l]++;
                // [r + 1, ru] 全删
                if (r + 1 < ru) Diff[r + 1]--, Diff[ru + 1]++;

                // v的子树全删
                l = lca.L[v], r = lca.R[v];
                Diff[l]--, Diff[r + 1]++;

            } else {
                int l = lca.L[u], r = lca.R[u];
                // u的子树全删
                Diff[l]--, Diff[r + 1]++;

                l = lca.L[v], r = lca.R[v];
                // v的子树全删
                Diff[l]--, Diff[r + 1]++;
            }
        }
    }

    int ans = 0;
    for (int i = 1; i <= n; i++) {
        Diff[i] += Diff[i - 1];
        if (Diff[i] == 0) ans++;
    }
    cout << ans << "\n";
}

int main() {
    ios;
    cout << fixed << setprecision(20);

    int T = 1; 
    // cin >> T;
    while (T--) {
    	solve();
    }
    return 0;
}
