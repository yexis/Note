// b:最大的v的二进制位数 例如v=8,b=3
int b;
struct Node {
    int val;
    vector<Node*> child;
    Node() {
        val = 0;
        child.resize(2);
    }
};
struct Trie {
    Node* root;
    Trie() {
        root = new Node();
    }
    void add(int v) {
        auto p = root;
        for (int i = b; i >= 0; i--) {
            int bit = v >> i & 1;
            if (p->child[bit] == nullptr) {
                p->child[bit] = new Node();
            }
            p = p->child[bit];
            p->val++;
        }
    }
    void del(int v) {
        auto p = root;
        for (int i = b; i >= 0; i--) {
            int bit = v >> i & 1;
            p = p->child[bit];
            p->val--;
        }
    }
    int ask(int v) {
        int ans = 0;
        auto p = root;
        for (int i = b; i >= 0; i--) {
            int bit = v >> i & 1;
            if (p->child[1 ^ bit] && p->child[1 ^ bit]->val > 0) {
                ans |= 1 << i;
                bit ^= 1;
            }
            p = p->child[bit];
        }
        return ans;
    }
};

##### 二叉Trie树

https://atcoder.jp/contests/abc475/tasks/abc475_e

> 一个集合，每次根据某个二元属性筛选出 0 类或 1 类，这像什么数据结构？答案是Trie
>
> 我们不是查询某个字符串是否存在，而是模拟一个动态筛选过程。
>
> Trie树口诀：
>
> 1. 字符串长度不大 
> 2. 每个字符串逐位处理 
> 3. 当前集合按照某一位分成两类 
> 4. 后续操作只在其中一类继续 
> 5. 动态修改某个字符串的某一位

ll power(ll x, ll b, ll m = mod) {
    ll ans = 1;
    while (b) {
        if (b & 1) {
            ans *= x;
            ans %= m;
        }
        x *= x;
        x %= m;
        b >>= 1;
    }
    return ans;

// 二叉Trie树模板开始
struct Node {
    int val;
    vector<Node*> child;

        val = 0;
        child.resize(2);
    }
};
struct Trie {
    Node* root;
    Trie() {
        root = new Node();
    }
    void add(vector<int>& masks) {
        int n = masks.size();
        auto p = root;
        for (int i = 0; i < n; i++) {
            int bit = masks[i];
            if (p->child[bit] == nullptr) {
                p->child[bit] = new Node();
            }
            p = p->child[bit];
            p->val++;
        }

    void del(vector<int>& masks) {
        int n = masks.size();
        auto p = root;

            int bit = masks[i];
            p = p->child[bit];
            p->val--;

    bool check(vector<int>& masks) {
        int n = masks.size(), rem = M;
        
        auto p = root;
        for (int i = 0; i < n; i++) {
            int corr = (p->child[1] != nullptr ? p->child[1]->val : 0);
            int bit = masks[i];

                p = p->child[1];
                if (p == nullptr) return false;

        }
        return false;
    }
};

void solve() {
    int n, m, K; cin >> n >> m >> K; 
    M = m;

    vector<string> S(n); for (int i = 0; i < n; i++) cin >> S[i];

    for (int i = 0; i < n; i++) for (int k = 0; k < K; k++) {
        if (S[i][k] == T[k]) bits[i][k] = 1;
        else bits[i][k] = 0;

        bool fg = trie.check(bits[i]); 
        if (fg) cout << "Yes" << "\n";
        else cout << "No" << "\n";
    }
}
