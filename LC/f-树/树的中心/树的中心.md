##### 3 key points

###### 3.1 什么是树的中心

使得距离该点最远的节点的距离最小。

> **什么是树的重心？**
>
> 删除该点后，剩余连通块的最大节点数最小。

 

##### 4 thought

###### 4.1 拓扑剥离法

如 5 code部分



###### 4.2 两端点BFS

第一次：任选一个节点$r$，使用BFS找到最远的节点$t$

第二次：从节点$t$出发，使用BFS找到最远的节点$s$

则$(s\rightarrow... \rightarrow t)$就是树的一个直径

void solve() {
    int n; cin >> n;
    vector<int> g[n], deg(n, 0);
    for (int i = 0; i < n - 1; i++) {
        int u, v; cin >> u >> v; u--, v--;
        g[u].push_back(v);
        g[v].push_back(u);
        deg[u]++, deg[v]++;
    }
    vector<int> colors(n); for (int i = 0; i < n; i++) cin >> colors[i];

    int step = 0, F = -1, G = -1;
    queue<int> q; for (int i = 0; i < n; i++) if (deg[i] == 1) q.push(i);
    while (q.size() > ) {
        int sz = q.size();
        for (int i = 0; i < sz; i++) {
            auto u = move(q.front()); q.pop();  
            G = F, F = u; // 如果是偶数，中心一定是最后一个
            for (auto v : g[u]) {
                if (--deg[v] == 1) {
                    q.push(v);
                }
            }
        }
        step++;
    }

    int diameter = (step << 1) - (q.size() == 0);

    // 奇数直径，队列最终剩余一个元素
    if (q.size()) {
        F = q.front(); q.pop();
        G = -1;
    }
    
    // debugs(F, G);

    if (diameter % 2 == 0) {
        // 偶数直径
        ll ans = 0;
        vector<int> mycolor(n + 1);
        unordered_map<int, int> tmp;
        auto dfs = [&](auto&& dfs, int u, int o, int d) -> void {
            if (d == (diameter >> 1)) {
                tmp[colors[u]]++;
                return;
            }
            for (auto& v : g[u]) if (v != o) {
                dfs(dfs, v, u, d + 1);
            }
        };
        
        for (auto& v : g[F]) {
            dfs(dfs, v, F, 1);
            for (auto& [fi, se] : tmp) {
                ans += 1ll * se * mycolor[fi];
                mycolor[fi] += tmp[fi];
            }
            
            tmp.clear();
        }
        cout << ans << "\n";
    } else {
        // 奇数直径
        auto dfs = [&](auto&& dfs, int u, int o, int d, vector<int>& mycolor) -> void {
            if (d == (diameter >> 1)) {
                mycolor[colors[u]]++;
                return;
            }
            for (auto& v : g[u]) if (v != o) {
                dfs(dfs, v, u, d + 1, mycolor);
            }
        };
        vector<int> colors1(n + 1); dfs(dfs, F, G, 0, colors1);
        vector<int> colors2(n + 1); dfs(dfs, G, F, 0, colors2);
        ll ans = 0;
        for (int i = 1; i <= n; i++) ans += 1ll * colors1[i] * colors2[i];
        cout << ans  << "\n";
    }
    
}

    	solve();
    }
    return 0;
}
