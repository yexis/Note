class Solution {
public:
    int minimumOR(vector<vector<int>>& g) {
        int m = g.size(), n = g[0].size();
        
        int mx = 0; for (int i = 0; i < m; i++) for (int j = 0; j < n; j++) mx = max(mx, g[i][j]);
        int M = 32 - __builtin_clz(mx);

        int ans = 0;
        for (int d = M - 1; d >= 0; d--) {
            int nxt = ans;
            bool ok1 = true;
            for (int i = 0; i < m; i++) {
                bool ok2 = false;
                for (int j = 0; j < n; j++) {
                    int x = (g[i][j] >> (d + 1));
                    int y = (nxt >> (d + 1));
                    if ((x & y) == x && !(g[i][j] >> d & 1) ) {
                        ok2 = true;
                        break;
                    }
                }
                if (!ok2) {
                    ok1 = false;
                }
            }
            if (!ok1) {
                ans |= 1 << d;
            }
        }
        return ans;
    }
};
